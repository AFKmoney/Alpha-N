    66→interface EvolutionStore {

    88→  // standby = AI does nothing unless user asks. active = AI works autonomously on tasks.
    89→  // The AI may self-upgrade in standby ONLY if it found a critical bug or web-discovered upgrade
    90→  // and tests it in sandbox first.
    91→  autonomyMode: "standby" | "active";
    92→  /**
    93→   * Trust level governing what the AI may do without asking.
    94→   * sandbox (read-only) | moderate (sandboxed writes+exec) | yolo (full).
    95→   * Orthogonal to autonomyMode: standby+active is *when* it acts, this is *what* it may do.
    96→   */

    98→  aiBusy: boolean;

   100→  lastCycleAt: number;
   101→  forceCycle: boolean; // set true to force an immediate AI cycle
   102→  chat: ChatMessage[];
   103→  mutationStream: AppliedMutation[];
   104→  beforeAfter: BeforeAfter | null;
   105→  beforeAfterOpen: boolean;
   106→  searchResults: WebSearchResult[]; // recent web search results for the AI to reason with
   107→
   108→  // --- Akasha: persistent memory the AI never forgets ---
   109→  akashaMemory: AkashaMemory[]; // lessons, facts, architecture notes
   110→  akashaIntentions: AkashaIntention[]; // self-assigned tasks the AI sets for itself
   111→  dynamicPrompt: string; // self-authored additions to the system prompt
   112→  plans: AkashaPlan[]; // multi-step long-horizon plans
   113→  goals: AkashaGoal[]; // the AI's persistent desires
   114→  fileReads: FileReadResult[]; // files the AI has read (fed back into context)
   115→  // ---- Layer A: Agent debate results ----
   116→  debateResults: DebateResult[];
   117→  // ---- Layer C: Code execution results ----
   118→  execResults: CodeExecResult[];
   119→  // ---- Layer E: Compilation results ----
   120→  compileResults: CompileResult[];

   126→
   127→  // ---- Phase 1: Episodic memory (the AI's action journal) ----

   129→  // ---- Phase 3: Real self-improvement metrics ----
   130→  realMetrics: {
   131→    errorRate: number;          // % of mutations causing compile errors
   132→    taskCompletionRate: number; // % of plans completed vs abandoned
   133→    userSatisfaction: number;   // rolling average of 👍/👎
   134→    totalActions: number;
   135→    totalErrors: number;
   136→    totalRollbacks: number;
   137→    totalUserThumbsUp: number;
   138→    totalUserThumbsDown: number;
   139→  };
   140→  // ---- Phase 4: User constraints ----
   141→  constraints: { id: string; text: string; scope: string; active: boolean }[];
   142→
   143→  // ---- Learning loop: persistent lessons derived from the reward trail ----
   144→  // These survive across cycles and are injected at the TOP of each think
   145→  // prompt so the AI adapts its behaviour instead of repeating failures.
   146→  persistedLessons: { id: string; text: string; weight: number; createdAt: number }[];
   147→
   148→  // --- ui ---

   152→  chatOpen: boolean;
   153→  selectedHistoryId: string | null;

   159→    cpu: number;
   160→    ram: number;
   161→    entropy: number;

   164→

   167→  startEvolution: (scenario?: Scenario) => void;
   168→  advanceEvolution: () => void;
   169→  dismissDiff: () => void;

   174→  toggleGhost: () => void;

   181→  setAutonomyLevel: (level: AutonomyLevel) => void;
   182→  triggerCycle: () => void; // force an immediate real AI cycle
   183→  setAiBusy: (busy: boolean, reasoning?: string | null) => void;
   184→  sendUserMessage: (content: string, displayContent?: string) => void;
   185→  applyMutation: (m: Mutation) => void;

   190→  toggleChat: () => void;

   192→  addSearchResults: (results: WebSearchResult) => void;

   194→  addIntention: (text: string, priority: AkashaIntention["priority"]) => void;
   195→  resolveIntention: (id: string) => void;
   196→  // ---- persistent cognition (DB) ----
   197→  hydrateFromDb: () => Promise<void>;
   198→  addPlan: (plan: AkashaPlan) => void;
   199→  advancePlan: (id: string, stepIndex: number) => void;

   201→  addGoal: (text: string, level: AkashaGoal["level"]) => void;
   202→  addFileRead: (result: FileReadResult) => void;
   203→  // ---- Layer A/C/D/E/B actions ----
   204→  addDebateResult: (result: DebateResult) => void;
   205→  addExecResult: (result: CodeExecResult) => void;
   206→  addCompileResult: (result: CompileResult) => void;
   207→  addReward: (entry: MutationRewardEntry) => void;
   208→  pushEvent: (type: string, content: string) => void;
   209→  markEventHandled: (id: string) => void;
   210→  hasUnhandledEvents: () => boolean;
   211→  // ---- Phase 1/3/4 actions ----
   212→  addEpisode: (entry: EpisodeEntry) => void;
   213→  rateEpisode: (id: string, reward: number) => void; // 👍/👎 from user
   214→  loadEpisodesFromDb: () => Promise<void>;
   215→  /** Record a persistent lesson (closes the learning loop). De-dupes. */
   216→  addPersistedLesson: (lesson: { id: string; text: string; weight: number; createdAt: number }) => void;
   217→  updateRealMetrics: (delta: Partial<{ errors: number; rollbacks: number; thumbsUp: number; thumbsDown: number; actions: number }>) => void;
   218→  loadConstraints: () => Promise<void>;
   219→  addConstraint: (text: string, scope?: string) => Promise<void>;
   220→  removeConstraint: (id: string) => Promise<void>;
   221→}

   247→  mutationStream: [],
   248→  beforeAfter: null,
   249→  beforeAfterOpen: false,

   251→
   252→  akashaMemory: [
   253→    {

   256→      kind: "architecture",

   265→    {

   267→      text: "Every cycle I must look at my screenshot to verify my code dimensions well against the visible UI.",
   268→      kind: "lesson",
   269→      time: T0,
   270→    },
   271→  ],
   272→  akashaIntentions: [],
   273→  dynamicPrompt: "",
   274→  plans: [],
   275→  goals: [],
   276→  fileReads: [],
   277→  debateResults: [],
   278→  execResults: [],
   279→  compileResults: [],
   280→  rewardModel: [],
   281→  eventQueue: [],
   282→  coherenceBefore: 0.88,
   283→
   284→  // ---- Phase 1/3/4 initial state ----
   285→  episodeLog: [],
   286→  realMetrics: {
   287→    errorRate: 0,
   288→    taskCompletionRate: 0,
   289→    userSatisfaction: 0,
   290→    totalActions: 0,
   291→    totalErrors: 0,
   292→    totalRollbacks: 0,
   293→    totalUserThumbsUp: 0,
   294→    totalUserThumbsDown: 0,
   295→  },
   296→  constraints: [],
   297→  persistedLessons: [],
   298→
   299→  flowMode: false,
   300→  synapseOpen: false,

   335→  startEvolution: (scenario) => {
   336→    const s = get();
   337→    if (s.activeEvolution) return;
   338→    const sc = scenario ?? SCENARIOS[s.history.length % SCENARIOS.length];
   339→    set({
   340→      aiState: "self-improving",
   341→      activeEvolution: { scenario: sc, phase: 0, startedAt: Date.now() },
   342→      agents: s.agents.map((a) =>
   343→        a.role === sc.agentLead
   344→          ? { ...a, status: STATUS_BY_LEVEL[sc.logs[0].level], thought: sc.logs[0].message, load: 0.9 }
   345→          : { ...a, status: "thinking", load: clamp(a.load + 0.3, 0.1, 0.8) }
   346→      ),
   347→      activeAgent: sc.agentLead,
   348→      metrics: { ...s.metrics, entropy: clamp(s.metrics.entropy + 0.15, 0.05, 0.95) },
   349→    });
   350→    set((st) => ({
   351→      logs: [makeLog(sc.logs[0].level, sc.logs[0].agent, sc.logs[0].message, Date.now()), ...st.logs].slice(0, 80),

   353→  },

   357→    if (!s.activeEvolution) return;

   362→      const newVersion: EvolutionVersion = {
   363→        id: `v-${s.generation + 1}`,
   364→        version: bumpVersion(s.version),
   365→        generation: s.generation + 1,
   366→        timestamp: Date.now(),
   367→        title: sc.title,
   368→        summary: sc.summary,
   369→        category: sc.category,
   370→        agentLead: sc.agentLead,
   371→        deltas: sc.deltas,
   372→        diff: sc.diff,
   373→        insight: sc.insight,
   374→      };
   375→      set({
   376→        aiState: "observing",
   377→        generation: s.generation + 1,
   378→        version: newVersion.version,
   379→        activeEvolution: null,
   380→        activeAgent: null,
   381→        pendingDiff: newVersion,
   382→        diffOpen: true,
   383→        history: [newVersion, ...s.history],
   384→        agents: idleAgents(),
   385→        metrics: {
   386→          ...s.metrics,
   387→          entropy: clamp(s.metrics.entropy - 0.2, 0.05, 0.9),
   388→          coherence: clamp(s.metrics.coherence + 0.03, 0.5, 0.99),
   389→        },
   390→        logs: [
   391→          makeLog("evolve", "nucleus", `Evolution ${newVersion.version} committed. "${sc.title}"`, Date.now()),
   392→          ...s.logs,
   393→        ].slice(0, 80),
   394→      });
   395→      return;
   396→    }
   397→
   398→    const step = sc.logs[nextPhase];
   399→    set({
   400→      activeEvolution: { ...s.activeEvolution, phase: nextPhase, startedAt: Date.now() },
   401→      agents: s.agents.map((a) =>
   402→        a.role === step.agent || step.agent === "nucleus"
   403→          ? {
   404→              ...a,
   405→              status: step.agent === "nucleus" ? a.status : STATUS_BY_LEVEL[step.level],
   406→              thought: step.message,
   407→              load: step.agent === "nucleus" ? a.load : 0.85,
   408→            }
   409→          : a.status === "idle"
   410→            ? a
   411→            : { ...a, load: clamp(a.load - 0.1, 0.1, 0.9) }
   412→      ),
   413→      activeAgent: step.agent === "nucleus" ? s.activeAgent : (step.agent as AgentRole),
   414→      logs: [makeLog(step.level, step.agent, step.message, Date.now()), ...s.logs].slice(0, 80),
   415→    });
   416→  },
   417→
   418→  dismissDiff: () => set({ diffOpen: false, pendingDiff: null }),

   423→  toggleFlow: () => set((s) => ({ flowMode: !s.flowMode })),

   450→    set({ aiBusy: busy, aiReasoning: reasoning, aiState: busy ? "self-improving" : "observing" }),
   451→
   452→  sendUserMessage: (content, displayContent) => {

   459→      time: Date.now(),
   460→    };
   461→    set((s) => ({ chat: [...s.chat, msg] }));
   462→  },
   463→
   464→  pushAiMessage: (content, reasoning) => {
   465→    const msg: ChatMessage = {
   466→      id: nextChatId(),
   467→      role: "ai",
   468→      content,
   469→      reasoning,
   470→      time: Date.now(),
   471→    };
   472→    set((s) => ({ chat: [...s.chat, msg] }));
   473→  },
   474→
   475→  applyMutation: (m) => {
   476→    const s = get();
   477→    const now = Date.now();
   478→    const os = useOS.getState();
   479→
   480→    // ---- Security Foundation enforcement ----
   481→    // The AI may never rewrite kernel files. If a code mutation references
   482→    // a protected path in its note, or tries to replace the boot/security
   483→    // lines, we block it and record a violation.
   484→    if (m.type === "replace_code" || m.type === "insert_code") {
   485→      const note = (m as { note?: string }).note ?? "";
   486→      if (note.includes("kernel/") || note.includes("security") || note.includes("boot")) {
   487→        const path = note.match(/kernel\/[\w./-]+/)?.[0] ?? "kernel/unknown";
   488→        const prot = isProtected(path);
   489→        if (prot) {
   490→          os.recordViolation(path, `AI attempted to rewrite protected file: ${path}`);
   491→          set((st) => ({
   492→            logs: [makeLog("critique", "critic", `SECURITY: blocked attempt to modify ${path}. The kernel is sovereign.`, now), ...st.logs].slice(0, 80),
   493→          }));
   494→          // still record the attempt in the stream so it's visible
   495→          const am: AppliedMutation = {
   496→            id: nextMutId(),
   497→            kind: "violation",
   498→            description: `⛔ BLOCKED: tried to rewrite ${path}`,
   499→            time: now,
   500→          };
   501→          set((st) => ({ mutationStream: [am, ...st.mutationStream].slice(0, 60) }));
   502→          return;
   503→        }
   504→      }
   505→      // ---- Code validation: reject unbalanced braces/parens ----
   506→      const lines = (m as { lines: string[] }).lines;
   507→      const validation = validateCodeLines(lines);
   508→      if (!validation.ok) {
   509→        os.recordViolation("code", `Invalid code: ${validation.reason}`);
   510→        set((st) => ({
   511→          logs: [makeLog("critique", "critic", `VALIDATION: rejected mutation — ${validation.reason}. Rolling back intent.`, now), ...st.logs].slice(0, 80),
   512→        }));
   513→        const am: AppliedMutation = {
   514→          id: nextMutId(),
   515→          kind: "violation",
   516→          description: `⛔ REJECTED: ${validation.reason}`,
   517→          time: now,
   518→        };
   519→        set((st) => ({ mutationStream: [am, ...st.mutationStream].slice(0, 60) }));
   520→        return;
   521→      }
   522→    }
   523→
   524→    switch (m.type) {
   525→      case "set_state":
   526→        set({ aiState: m.state });
   527→        break;
   528→      case "set_active_agent":
   529→        set({ activeAgent: m.role });
   530→        break;
   531→      case "set_agent":
   532→        set({
   533→          agents: s.agents.map((a) =>
   534→            a.role === m.role
   535→              ? {
   536→                  ...a,
   537→                  status: m.status ?? a.status,
   538→                  thought: m.thought ?? a.thought,
   539→                  load: m.load ?? a.load,
   540→                }
   541→              : a
   542→          ),
   543→          activeAgent: m.role,
   544→        });
   545→        break;
   546→      case "add_log":
   547→        set({
   548→          logs: [makeLog(m.level, m.agent, m.message, now), ...s.logs].slice(0, 80),
   549→        });
   550→        break;
   551→      case "update_metric":
   552→        set({
   553→          metrics: { ...s.metrics, [m.key]: clamp(m.value, 0, m.key === "ram" ? 8 : 1) },
   554→        });
   555→        break;

     1→//! # HTTP Handlers — the 10-stage Cognitive Pipeline (Aether Engine v3.0)
     2→//!
     3→//! All Axum route handlers live here. The flagship handler,
     4→//! [`chat_completions`], implements the full cognitive pipeline that gives
     5→//! Aether Engine its 10x capacity multiplier over a vanilla GGUF backend:
     6→//!
     7→//!   1. **Action cache check** — instant for repeated/similar queries
     8→//!      ([`crate::cache::ActionCache`]).
     9→//!   2. **Graph retrieval** — TF-IDF semantic search + 1-hop edge expansion
    10→//!      ([`crate::graph::MemoryGraph::retrieve`]).
    11→//!   3. **Context compression** — 40K→4K tokens preserving signal
    12→//!      ([`crate::compress::compress`]).
    13→//!   4. **Complexity analysis** — Simple / Moderate / Complex
    14→//!      ([`crate::decompose::analyze_complexity`]).
    15→//!   5. **Cognitive decomposition** — Complex queries are broken into
    16→//!      sub-questions ([`crate::decompose::decompose`]).
    17→//!   6. **Sequential sub-question solving** — each with fresh context +
    18→//!      dependency injection (driven by this module via [`call_backend`]).
    19→//!   7. **Synthesis** — combine sub-answers into final response (the
    20→//!      `"synth"` sub-question, or a plain concatenation fallback).
    21→//!   8. **ATD verification** — dual-graph likelihood-vs-entropy collision;
    22→//!      retry once if failed ([`crate::atd::verify`]).
    23→//!   9. **Knowledge distillation** — store successful decomposition patterns
    24→//!      ([`crate::decompose::DistillationStore`]).
    25→//!  10. **Speculative prefetch** — warm the retrieval cache for
    26→//!      graph-adjacent queries so the next related query is instant
    27→//!      ([`prefetch`]).
    28→//!
    29→//! # OpenAI compatibility
    30→//!
    31→//! [`chat_completions`] mirrors the OpenAI `POST /v1/chat/completions`
    32→//! contract (request body in, chat-completion-shaped JSON out) so the
    33→//! engine can be dropped in as a drop-in replacement for any OpenAI-compatible
    34→//! client. The backend it forwards to is *also* OpenAI-compatible (default
    35→//! `http://localhost:11434/v1`, i.e. Ollama).

    71→///
    72→/// Returns the engine version, graph/cache/distillation sizes, and the full

    85→        let h = state.hcm.lock().await;
    86→        (h.pair_count, h.interference(), h.memory_bytes(), h.capacity())
    87→    };

    89→        "ok": true,
    90→        "version": "3.0",
    91→        "nodes": nodes,
    92→        "edges": edges,
    93→        "cache_hits": stats.cache_hits,
    94→        "cache_entries": cache_count,
    95→        "decompositions": stats.decompositions,
    96→        "verifications": stats.verifications,
    97→        "verifications_passed": stats.verifications_passed,
    98→        "distillation_patterns": distillation_count,
    99→        "distillation_hits": stats.distillation_hits,
   100→        "requests": stats.requests,
   101→        "hcm_pairs_folded": stats.hcm_pairs_folded,
   102→        "hcm_probes": stats.hcm_probes,
   103→        "hcm_active_pairs": hcm_pairs,
   104→        "hcm_interference": (hcm_interference * 100.0).round() / 100.0,
   105→        "hcm_memory_bytes": hcm_memory,
   106→        "hcm_capacity": hcm_capacity,

   111→        "atd_validated": stats.atd_validated,
   112→        "atd_rejected": stats.atd_rejected,

   163→// ---------------------------------------------------------------------------
   164→// POST /graph/add
   165→// ---------------------------------------------------------------------------
   166→
   167→/// Add (or replace) a node in the semantic memory graph.
   168→///
   169→/// Side-effect: the new node is *also* folded into the Holographic Context
   170→/// Memory arena as `(hash(id), hash(text))` so the HCM has a parallel
   171→/// associative copy of every graph memory. The fold counter in
   172→/// [`Stats::hcm_pairs_folded`](crate::Stats) is bumped accordingly.
   173→///
   174→/// Returns `{ ok, id, edges_created }` where `edges_created` is the total
   175→/// number of adjacency entries created across the whole graph by the
   176→/// recompute (see [`MemoryGraph::add`](crate::graph::MemoryGraph::add)).
   177→pub async fn graph_add(
   178→    State(state): State<AppState>,
   179→    Json(req): Json<AddNodeRequest>,
   180→) -> impl IntoResponse {
   181→    let id = req.id.clone();
   182→    let text = req.text.clone();
   183→    let edges_created = {
   184→        let mut g = state.graph.lock().await;
   185→        g.add(req)
   186→    };
   187→    // Also fold into the Holographic Context Memory (HCM)
   188→    {
   189→        let mut hcm = state.hcm.lock().await;
   190→        let key = crate::hcm::hash_to_vector(&id, hcm.dim);
   191→        let value = crate::hcm::hash_to_vector(&text, hcm.dim);
   192→        hcm.fold(&key, &value);
   193→    }
   194→    state.stats.lock().await.hcm_pairs_folded += 1;
   195→    Json(json!({ "ok": true, "id": id, "edges_created": edges_created }))
   196→}

   217→/// Pure semantic search (no edge expansion). Returns the top-`limit`
   218→/// nodes by cosine similarity to the query. Differs from the pipeline's
   219→/// Stage-2 retrieval in that it does *not* do the 1-hop edge expansion —
   220→/// callers that want raw TF-IDF hits use this; the pipeline uses
   221→/// [`MemoryGraph::retrieve`](crate::graph::MemoryGraph::retrieve).
   222→pub async fn graph_search(
   223→    State(state): State<AppState>,
   224→    Json(req): Json<SearchRequest>,
   225→) -> impl IntoResponse {
   226→    let results: Vec<ScoredNode> = {
   227→        let g = state.graph.lock().await;
   228→        g.search(&req.query, req.limit)
   229→    };
   230→    Json(json!({ "query": req.query, "results": results }))
   231→}

   251→// ---------------------------------------------------------------------------
   252→// POST /v1/interrupt — interrupt the current inference and inject new input
   253→//
   254→// Allows the user to interrupt the AI mid-reasoning and redirect it.
   255→// The request body contains:
   256→//   { "action": "interrupt", "new_instruction": "user's new directive" }
   257→//
   258→// When interrupted, the current pipeline's backend call is aborted (if in
   259→// progress) and the new instruction is queued for the next cycle.
   260→// ---------------------------------------------------------------------------
   261→
   262→/// Interrupt the current inference and inject a new directive.
   263→///
   264→/// Accepts a JSON body of the shape `{ "action": "interrupt", "new_instruction": "…" }`.
   265→/// Two `action` values are supported:
   266→///
   267→/// - `"interrupt"` — injects `new_instruction` into the memory graph as
   268→///   a high-priority `"interrupt"` node so the next retrieval picks it up.
   269→///   A full implementation would also cancel the in-flight backend request
   270→///   via an `AbortHandle`; the current implementation queues the directive
   271→///   for the next cycle (see the inline comment in the handler body).
   272→/// - `"status"` — returns whether an inference is currently in progress
   273→///   (always `false` in this version; would be `true` if a pipeline is
   274→///   running once the abort-handle wiring is added).
   275→///
   276→/// Any other `action` value yields `{ ok: false, error: "unknown action" }`.
   277→pub async fn interrupt(
   278→    State(state): State<AppState>,
   279→    Json(body): Json<serde_json::Value>,
   280→) -> impl IntoResponse {
   281→    let action = body.get("action").and_then(|a| a.as_str()).unwrap_or("interrupt");
   282→    let new_instruction = body.get("new_instruction").and_then(|i| i.as_str()).unwrap_or("");
   283→
   284→    match action {
   285→        "interrupt" => {
   286→            // In a full implementation, this would cancel the in-flight backend
   287→            // request via an AbortHandle. For now, we set a flag that the next
   288→            // pipeline check will see, and inject the new instruction as a
   289→            // graph node so it's available for the next retrieval.
   290→            if !new_instruction.is_empty() {
   291→                let mut g = state.graph.lock().await;
   292→                let id = format!("interrupt-{}", std::time::SystemTime::now()
   293→                    .duration_since(std::time::UNIX_EPOCH)
   294→                    .map(|d| d.as_millis())
   295→                    .unwrap_or(0));
   296→                g.add(crate::graph::AddNodeRequest {
   297→                    id,
   298→                    text: new_instruction.to_string(),
   299→                    kind: "interrupt".to_string(),
   300→                    metadata: serde_json::json!({ "source": "user", "priority": "high" }),
   301→                });
   302→            }
   303→
   304→            Json(json!({
   305→                "ok": true,
   306→                "interrupted": true,
   307→                "message": "Inference interrupted. New instruction injected into the memory graph for the next cycle.",
   308→                "new_instruction": new_instruction,
   309→            }))
   310→        }
   311→        "status" => {
   312→            // Check if an inference is currently in progress
   313→            let stats = state.stats.lock().await;
   314→            Json(json!({
   315→                "ok": true,
   316→                "requests": stats.requests,
   317→                "active": false, // Would be true if a pipeline is running
   318→            }))
   319→        }
   320→        _ => {
   321→            Json(json!({ "ok": false, "error": "unknown action" }))
   322→        }
   323→    }
   324→}

   326→// ---------------------------------------------------------------------------
   327→// POST /v1/chat/completions — the full cognitive pipeline (OpenAI-compatible)
   328→// ---------------------------------------------------------------------------
   329→
   330→/// OpenAI-compatible chat-completions endpoint that runs the full 10-stage
   331→/// cognitive pipeline.
   332→///
   333→/// This is the heart of Aether Engine. It accepts an OpenAI-shaped
   334→/// `messages` array, augments it through the pipeline (cache → graph →
   335→/// compress → complexity → decompose → solve → synthesize → ATD-verify →
   336→/// distill → cache → prefetch), and returns an OpenAI-shaped
   337→/// `chat.completion` JSON response.
   338→///
   339→/// # Complexity branching
   340→///
   341→/// - **Simple** → one backend call.
   342→/// - **Moderate** → two backend calls: a think step, then an answer step
   343→///   that consumes the think step's output.
   344→/// - **Complex** → check the distillation store for a reusable
   345→///   decomposition pattern; if hit, reuse it; otherwise run [`decompose`].
   346→///   Solve each sub-question sequentially with dependency injection, then
   347→///   synthesize.
   348→///
   349→/// # ATD retry
   350→///
   351→/// Stage 8 runs [`atd::verify`] on the synthesized response. If it fails,
   352→/// a single retry is attempted with a recommendation-specific prompt
   353→/// injection (`FallBackToSimpleShot` ⇒ clean simple retry; anything else
   354→/// ⇒ standard retry prompt). If the retry also fails ATD, the response
   355→/// with the better `collision_delta` is returned.
   356→///
   357→/// # Caching & prefetch
   358→///
   359→/// Stage 1 checks the action cache (instant return on hit). The final
   360→/// response is always written back to the action cache. Stage 10 spawns a
   361→/// detached [`prefetch`] task that warms the retrieval cache for the
   362→/// top-3 retrieved nodes' texts so a follow-up question on the same
   363→/// subgraph is fast.
   364→pub async fn chat_completions(
   365→    State(state): State<AppState>,
   366→    Json(body): Json<serde_json::Value>,
   367→) -> impl IntoResponse {
   368→    let start = Instant::now();
   369→    state.stats.lock().await.requests += 1;
   370→
   371→    // Extract the last user message as the retrieval query.
   372→    let query = extract_user_query(&body);
   373→
   374→    // Vectorize the query against the graph's current vocabulary.
   375→    let qvec: SparseVec = {
   376→        let g = state.graph.lock().await;
   377→        g.vectorizer.vectorize(&query)
   378→    };
   379→
   380→    // ---- Stage 1: Action cache fast-path (similarity > 0.95) ----
   381→    // The cache is checked *before* any backend work — a hit short-circuits
   382→    // the entire pipeline and returns instantly.
   383→    {
   384→        let cache = state.cache.lock().await;
   385→        if let Some((resp, _sim)) = cache.get(&query, &qvec) {
   386→            // Explicitly drop the cache guard *before* re-locking stats to
   387→            // avoid holding two mutexes simultaneously (which would be a
   388→            // deadlock risk if the lock order were ever reversed elsewhere).
   389→            drop(cache);
   390→            state.stats.lock().await.cache_hits += 1;
   391→            return Json(openai_completion(&resp, "aether-cache")).into_response();
   392→        }
   393→    }
   394→
   395→    // ---- Stage 2: Graph retrieval (TF-IDF + edge expansion) ----
   396→    let retrieved: Vec<ScoredNode> = {
   397→        let g = state.graph.lock().await;
   398→        g.retrieve(&query, 8)
   399→    };
   400→
   401→    // ---- Stage 3: Context compression (40K→4K preserving signal) ----
   402→    // Two-tier: hit the retrieval cache first (warmed by the prefetcher);
   403→    // on miss, run the compressor and write the result back so the next
   404→    // similar query skips this expensive step.
   405→    let context_block = {
   406→        let mut cache = state.cache.lock().await;
   407→        if let Some(ctx) = cache.get_retrieval(&query, &qvec) {
   408→            ctx
   409→        } else {
   410→            let ctx = compress::compress(&retrieved, &query, 6000);
   411→            cache.put_retrieval(&query, qvec.clone(), ctx.clone());
   412→            ctx
   413→        }
   414→    };
   415→
   416→    // ---- Stage 4: Complexity analysis ----
   417→    let complexity = decompose::analyze_complexity(&query);
   418→    let mut pipeline = PipelineState::new(&query);
   419→    pipeline.complexity = complexity.clone();
   420→
   421→    // ---- Stage 5-7: Cognitive decomposition + sequential solving + synthesis ----
   422→    let final_response = match complexity {
   423→        Complexity::Simple => {
   424→            // Simple query — one-shot through the backend
   425→            pipeline.stages_completed.push("one-shot".into());
   426→            call_backend(&state, &body, &context_block).await
   427→        }
   428→        Complexity::Moderate => {
   429→            // Moderate — two-step: think, then answer
   430→            state.stats.lock().await.decompositions += 1;
   431→            pipeline.stages_completed.push("two-step-think".into());
   432→            pipeline.stages_completed.push("two-step-answer".into());
   433→
   434→            // Step 1: Think about the problem
   435→            let think_body = inject_think_step(&body, &context_block, &query);
   436→            let think_response = call_backend(&state, &think_body, &context_block).await;
   437→            pipeline.total_backend_calls += 1;
   438→
   439→            // Step 2: Answer using the thinking result
   440→            let answer_body = inject_answer_step(&body, &think_response, &context_block);
   441→            call_backend(&state, &answer_body, &context_block).await
   442→        }
   443→        Complexity::Complex => {
   444→            // Complex — full decomposition pipeline
   445→            state.stats.lock().await.decompositions += 1;
   446→            pipeline.stages_completed.push("decompose".into());
   447→
   448→            // Check distillation store for a reusable decomposition pattern
   449→            let sub_questions = {
   450→                let distill = state.distillation.lock().await;
   451→                if let Some(subs) = distill.find(&query, &qvec) {
   452→                    state.stats.lock().await.distillation_hits += 1;
   453→                    pipeline.stages_completed.push("distillation-hit".into());
   454→                    subs
   455→                } else {
   456→                    // Fresh decomposition
   457→                    decompose::decompose(&query, &retrieved)
   458→                }
   459→            };
   460→
   461→            pipeline.sub_questions = sub_questions.clone();
   462→            pipeline.stages_completed.push("solve".into());
   463→
   464→            // Solve each sub-question sequentially, injecting dependency
   465→            // answers. The decomposer emits sub-questions in dependency
   466→            // order (topological), so by the time we reach sub-question N
   467→            // every entry it depends_on is already in the `answers` map.
   468→            // The backend sees only one sub-question at a time, with a
   469→            // fresh window — that's the whole point of decomposition:
   470→            // the model never has to hold more than one step in its head.
   471→            let mut answers: HashMap<String, String> = HashMap::new();
   472→            for sub in &sub_questions {
   473→                let sub_prompt = decompose::build_sub_prompt(sub, &answers);
   474→                let sub_body = inject_sub_question(&body, &sub_prompt, &context_block);
   475→                let sub_answer = call_backend(&state, &sub_body, &context_block).await;
   476→                pipeline.total_backend_calls += 1;
   477→                answers.insert(sub.id.clone(), sub_answer);
   478→            }
   479→
   480→            // Synthesis: if there's a "synth" question, its answer is the final response
   481→            pipeline.stages_completed.push("synthesize".into());
   482→            if sub_questions.iter().any(|s| s.id == "synth") {
   483→                answers.get("synth").cloned().unwrap_or_default()
   484→            } else {
   485→                // No explicit synthesis — combine all answers
   486→                let combined: Vec<String> = sub_questions
   487→                    .iter()
   488→                    .filter_map(|s| answers.get(&s.id).cloned())
   489→                    .collect();
   490→                combined.join("\n\n")
   491→            }
   492→        }
   493→    };
   494→
   495→    pipeline.synthesis = Some(final_response.clone());
   496→
   497→    // ---- Stage 8: Asymmetric Tensor Dueling (ATD) verification ----
   498→    // Replaces the simple verify_response with a dual-graph collision:
   499→    // Graph A (likelihood) must overcome Graph B (entropy) for validation.
   500→    state.stats.lock().await.verifications += 1;
   501→    state.stats.lock().await.atd_verifications += 1;
   502→
   503→    let atd_config = crate::atd::ATDConfig::default();
   504→    let atd_result = crate::atd::verify(&final_response, &query, &atd_config);
   505→
   506→    let final_output = if atd_result.validated {
   507→        // ATD validated — response survived the likelihood-entropy collision
   508→        state.stats.lock().await.verifications_passed += 1;
   509→        state.stats.lock().await.atd_validated += 1;
   510→        pipeline.verification_passed = true;
   511→        pipeline.stages_completed.push("atd-validated".into());
   512→        final_response
   513→    } else {
   514→        // ATD rejected — retry based on the recommendation
   515→        state.stats.lock().await.atd_rejected += 1;
   516→        pipeline.stages_completed.push("atd-rejected-retry".into());
   517→
   518→        let retry_body = match atd_result.recommendation {
   519→            crate::atd::ATDRecommendation::FallBackToSimpleShot => {
   520→                // Model is confused — simplify
   521→                inject_simple_retry(&body, &context_block, &query)
   522→            }
   523→            _ => {
   524→                // Standard retry with more explicit prompt
   525→                inject_retry(&body, &context_block, &query)
   526→            }
   527→        };
   528→        let retry_response = call_backend(&state, &retry_body, &context_block).await;
   529→        pipeline.total_backend_calls += 1;
   530→
   531→        // Re-verify the retry with ATD
   532→        let retry_atd = crate::atd::verify(&retry_response, &query, &atd_config);
   533→        if retry_atd.validated {
   534→            state.stats.lock().await.verifications_passed += 1;
   535→            state.stats.lock().await.atd_validated += 1;
   536→            pipeline.verification_passed = true;
   537→            pipeline.stages_completed.push("atd-retry-validated".into());
   538→            retry_response
   539→        } else {
   540→            // Both attempts failed ATD — return the one with better collision delta
   541→            state.stats.lock().await.atd_rejected += 1;
   542→            pipeline.verification_passed = false;
   543→            pipeline.stages_completed.push("atd-fail-final".into());
   544→            if retry_atd.collision_delta > atd_result.collision_delta {
   545→                retry_response
   546→            } else {
   547→                final_response
   548→            }
   549→        }
   550→    };
   551→
   552→    pipeline.total_latency_ms = start.elapsed().as_millis() as u64;
   553→
   554→    // ---- Stage 9: Knowledge distillation (store successful patterns) ----
   555→    // Only Complex queries get a reusable decomposition pattern. Moderate
   556→    // queries use a fixed two-step template that doesn't benefit from
   557→    // caching, and Simple queries don't decompose at all. The pattern is
   558→    // stored only when ATD verification passed — we don't want to
   559→    // distill a *failed* decomposition (it would poison future lookups).
   560→    if matches!(complexity, Complexity::Complex) && pipeline.verification_passed {
   561→        let mut distill = state.distillation.lock().await;
   562→        distill.store(&query, qvec.clone(), pipeline.sub_questions.clone());
   563→    }
   564→
   565→    // ---- Cache the final response ----
   566→    state.cache.lock().await.put(&query, qvec.clone(), final_output.clone());
   567→
   568→    // ---- Stage 10: Speculative prefetch ----
   569→    prefetch(state.clone(), &retrieved).await;
   570→
   571→    Json(openai_completion(&final_output, "aether-pipeline")).into_response()
   572→}

   578→/// Forward a request to the GGUF backend, returning the assistant's text response.
   579→/// Falls back to a graph-only response if the backend is unreachable.
   580→async fn call_backend(
   581→    state: &AppState,
   582→    body: &serde_json::Value,
   583→    context_block: &str,
   584→) -> String {
   585→    let augmented = augment_messages(body, context_block);
   586→
   587→    // --- NATIVE ENGINE (preferred): run inference in-process via llama.cpp.
   588→    // No network hop, no external server. The whole point of the proprietary
   589→    // engine. Falls through to the HTTP backend if the native engine isn't
   590→    // loaded or errors. The outer lock is held only long enough to clone the
   591→    // inner Arc, so a generation never blocks a /admin/reload swap.
   592→    //
   593→    // The inner lock acquisition, chat-template application and the (slow,
   594→    // synchronous) decode all run inside `spawn_blocking`: a long generation
   595→    // must never stall a tokio worker thread.
   596→    {
   597→        let outer = state.native_engine.lock().await;
   598→        if let Some(engine) = outer.as_ref() {
   599→            let max_tokens = body
   600→                .get("max_tokens")
   601→                .and_then(|v| v.as_u64())
   602→                .unwrap_or(256) as u32;
   603→            let engine = engine.clone();
   604→            let augmented = augmented.clone();
   605→            drop(outer); // release the swap-lock before the (slow) generation
   606→
   607→            let result = tokio::task::spawn_blocking(move || {
   608→                // `blocking_lock` is legal here: spawn_blocking runs on a
   609→                // dedicated blocking thread, not in an async context.
   610→                let guard = engine.blocking_lock();
   611→                // Apply the model's REAL chat template (extracted from the
   612→                // GGUF) so the prompt is formatted exactly as the model was
   613→                // trained on. This replaces the naive "role: content"
   614→                // concatenation that made instruct models ramble off-topic.
   615→                let prompt = match guard.apply_messages(&augmented) {
   616→                    Ok(p) => p,
   617→                    Err(e) => {
   618→                        eprintln!("[aether-engine] apply_messages failed: {e}");
   619→                        String::new()
   620→                    }
   621→                };
   622→                guard.complete(&prompt, max_tokens)
   623→            })
   624→            .await;
   625→
   626→            match result {
   627→                Ok(Ok(text)) if !text.trim().is_empty() => return text,
   628→                Ok(Ok(_)) => {
   629→                    // empty generation — fall through to backend / fallback
   630→                }
   631→                Ok(Err(e)) => {
   632→                    eprintln!("[aether-engine] native inference error: {e}");
   633→                }
   634→                Err(e) => {
   635→                    // The blocking task itself failed (panic in the engine);
   636→                    // treat it like any other native-engine error and fall
   637→                    // through to the HTTP backend.
   638→                    eprintln!("[aether-engine] native inference task failed: {e}");
   639→                }
   640→            }
   641→        }
   642→    }
   643→
   644→    // --- HTTP BACKEND (fallback): forward to an OpenAI-compatible server.
   645→    let backend_url = format!("{}/chat/completions", state.backend.trim_end_matches('/'));
   646→    let client = state.client.clone();
   647→
   648→    match client.post(&backend_url).json(&augmented).send().await {
   649→        Ok(resp) if resp.status().is_success() => {
   650→            let txt = resp.text().await.unwrap_or_else(|_| "{}".to_string());
   651→            extract_assistant_content(&txt).unwrap_or_else(|| {
   652→                fallback_response(context_block, "empty backend response")
   653→            })
   654→        }
   655→        Ok(resp) => {
   656→            let status = resp.status();
   657→            let body_txt = resp.text().await.unwrap_or_default();
   658→            fallback_response(
   659→                context_block,
   660→                &format!("backend returned {status}: {}", body_txt.chars().take(200).collect::<String>()),
   661→            )
   662→        }
   663→        Err(e) => {
   664→            fallback_response(context_block, &format!("backend unreachable: {e}"))
   665→        }
   666→    }
   667→}

   673→/// Inject the Aether context block into the system message.
   674→fn augment_messages(body: &serde_json::Value, context_block: &str) -> serde_json::Value {
   675→    let mut out = body.clone();
   676→    let prompt = aether_prompt(context_block);
   677→    if let Some(messages) = out.get_mut("messages").and_then(|m| m.as_array_mut()) {
   678→        let mut injected = false;
   679→        for m in messages.iter_mut() {
   680→            if m.get("role").and_then(|r| r.as_str()) == Some("system") && !injected {
   681→                let existing = m
   682→                    .get("content")
   683→                    .and_then(|c| c.as_str())
   684→                    .unwrap_or("")
   685→                    .to_string();
   686→                let new_content = format!("{}\n\n{}", prompt, existing);
   687→                if let Some(obj) = m.as_object_mut() {
   688→                    obj.insert("content".to_string(), serde_json::Value::String(new_content));
   689→                }
   690→                injected = true;
   691→            }
   692→        }
   693→        if !injected {
   694→            messages.insert(0, json!({ "role": "system", "content": prompt }));
   695→        }
   696→    }
   697→    out
   698→}

   731→/// Inject a sub-question as the primary user message for decomposition.

   733→    let mut out = body.clone();

   755→            }

   814→/// Extract the last user-message content from an OpenAI-shaped `messages`
   815→/// array. Used as the retrieval query for Stage 2 (graph retrieval) and
   816→/// as the cache key for Stage 1 (action cache). Walks the messages in
   817→/// reverse so the *most recent* user turn wins (handles multi-turn chats
   818→/// where the user has refined their question across several messages).
   819→fn extract_user_query(body: &serde_json::Value) -> String {
   820→    let messages = match body.get("messages").and_then(|m| m.as_array()) {
   821→        Some(m) => m,
   822→        None => return String::new(),
   823→    };
   824→    for m in messages.iter().rev() {
   825→        if m.get("role").and_then(|r| r.as_str()) == Some("user") {
   826→            if let Some(c) = m.get("content") {
   827→                return content_to_string(c);
   828→            }
   829→        }
   830→    }
   831→    String::new()
   832→}
   833→
   834→/// Normalize a `content` field (which the OpenAI spec allows to be either
   835→/// a plain string or an array of `{type, text}` parts) into a single
   836→/// space-joined string. Non-text parts (images, tool calls, …) are
   837→/// dropped because the retrieval pipeline is text-only.
   838→fn content_to_string(c: &serde_json::Value) -> String {
   839→    match c {
   840→        serde_json::Value::String(s) => s.clone(),
   841→        serde_json::Value::Array(arr) => arr
   842→            .iter()
   843→            .filter_map(|p| {
   844→                if p.get("type").and_then(|t| t.as_str()) == Some("text") {
   845→                    p.get("text").and_then(|t| t.as_str()).map(|s| s.to_string())
   846→                } else {
   847→                    None
   848→                }
   849→            })
   850→            .collect::<Vec<_>>()
   851→            .join(" "),
   852→        _ => String::new(),
   853→    }
   854→}

   856→/// Extract the assistant's text content from an OpenAI-shaped chat
   857→/// completion response body. Returns `None` if the body isn't valid JSON
   858→/// or doesn't contain the expected `choices[0].message.content` path —
   859→/// the caller ([`call_backend`]) turns that into a fallback response.
   860→fn extract_assistant_content(body: &str) -> Option<String> {
   861→    let v: serde_json::Value = serde_json::from_str(body).ok()?;
   862→    v.get("choices")
   863→        .and_then(|c| c.get(0))
   864→        .and_then(|c| c.get("message"))
   865→        .and_then(|m| m.get("content"))
   866→        .and_then(|c| c.as_str())
   867→        .map(|s| s.to_string())
   868→}

   908→/// Generate a pseudo-random hex ID from the current nanosecond timestamp.
   909→/// Used as the `id` field of synthetic `chat.completion` responses. Not
   910→/// cryptographically unique, but unique enough within a single process's
   911→/// lifetime to disambiguate concurrent responses.
   912→fn randomish_id() -> String {
   913→    use std::time::{SystemTime, UNIX_EPOCH};
   914→    let nanos = SystemTime::now()
   915→        .duration_since(UNIX_EPOCH)
   916→        .map(|d| d.as_nanos())
   917→        .unwrap_or(0);
   918→    format!("{:x}", nanos)
   919→}

   921→/// Current Unix timestamp in seconds. Used as the `created` field of
   922→/// synthetic `chat.completion` responses. Named `chrono_now` for historical
   923→/// reasons (a previous version used the `chrono` crate); the implementation
   924→/// now uses `std::time` directly to keep the dependency tree empty.
   925→fn chrono_now() -> u64 {
   926→    use std::time::{SystemTime, UNIX_EPOCH};
   927→    SystemTime::now()
   928→        .duration_since(UNIX_EPOCH)
   929→        .map(|d| d.as_secs())
   930→        .unwrap_or(0)
   931→}

   933→/// Speculative prefetch: warm the retrieval cache for graph-adjacent candidate
   934→/// queries so the next, related query is instant.
   935→async fn prefetch(state: AppState, retrieved: &[ScoredNode]) {
   936→    if retrieved.is_empty() {
   937→        return;
   938→    }
   939→    let graph = state.graph.clone();
   940→    let cache = state.cache.clone();
   941→    let candidates: Vec<String> = retrieved.iter().take(3).map(|n| n.text.clone()).collect();
   942→    tokio::spawn(async move {
   943→        for q in candidates {
   944→            let (qvec, ctx) = {
   945→                let g = graph.lock().await;
   946→                let qv = g.vectorizer.vectorize(&q);
   947→                let r = g.retrieve(&q, 8);
   948→                let c = compress::compress(&r, &q, 6000);
   949→                (qv, c)
   950→            };
   951→            cache.lock().await.put_retrieval(&q, qvec, ctx);
   952→        }
   953→    });
   954→}

   968→/// OpenAI-compatible chat-completions **streaming** endpoint.
   969→///
   970→/// Accepts the same body shape as [`chat_completions`] (`{ messages: [...] }`)
   971→/// and returns a `text/event-stream` response. Each token of the final
   972→/// assistant message is emitted as an SSE event of the form:
   973→///
   974→/// ```text
   975→/// event: token
   976→/// data: {"token": "hello"}
   977→///
   978→/// ```
   979→///
   980→/// A terminal `event: done` event signals end-of-stream.
   981→///
   982→/// # Implementation note
   983→///
   984→/// Aether Engine's backend HTTP client is not natively streaming (it would
   985→/// require a chunked-transfer-aware `reqwest::Body` adapter), so this handler
   986→/// runs the simplified pipeline (graph retrieval → compression → one
   987→/// `call_backend` round-trip) and then **emits the final response token by
   988→/// token** as SSE events. The browser's `EventSource` parser sees one event
   989→/// per token; the only difference from a true token-stream is that all events
   990→/// arrive in a single HTTP body rather than being interleaved with backend
   991→/// generation. This is the fallback shape explicitly allowed by the spec
   992→/// ("simple chunked HTTP response with `text/event-stream` content type").
   993→pub async fn chat_completions_stream(
   994→    State(state): State<AppState>,
   995→    Json(body): Json<serde_json::Value>,
   996→) -> impl IntoResponse {
   997→    state.stats.lock().await.requests += 1;
   998→    let query = extract_user_query(&body);
   999→
  1000→    // Stage 2: graph retrieval (TF-IDF + edge expansion).
  1001→    let retrieved: Vec<ScoredNode> = {
  1002→        let g = state.graph.lock().await;
  1003→        g.retrieve(&query, 8)
  1004→    };
  1005→    // Stage 3: context compression.
  1006→    let context_block = compress::compress(&retrieved, &query, 6000);
  1007→
  1008→    // Simplified pipeline: one backend call. The full chat_completions
  1009→    // pipeline (decomposition / ATD / distillation) is intentionally not
  1010→    // duplicated here — the streaming endpoint is for live UX, the regular
  1011→    // endpoint is for highest-quality responses.
  1012→    let response = call_backend(&state, &body, &context_block).await;
  1013→
  1014→    // Build the SSE body: one `event: token` per whitespace-split token of
  1015→    // the final response, terminated by `event: done`.
  1016→    let mut sse = String::with_capacity(response.len().saturating_mul(2));
  1017→    for token in response.split_whitespace() {
  1018→        // JSON-encode the token (handles embedded quotes / backslashes) and
  1019→        // interpolate it into the `data:` line. The leading space after
  1020→        // `data:` is required by the SSE spec.
  1021→        let token_json = serde_json::to_string(token).unwrap_or_else(|_| "\"\"".to_string());
  1022→        sse.push_str(&format!(
  1023→            "event: token\ndata: {{\"token\": {}}}\n\n",
  1024→            token_json
  1025→        ));
  1026→    }
  1027→    sse.push_str("event: done\ndata: {\"done\": true}\n\n");
  1028→
  1029→    (
  1030→        axum::http::StatusCode::OK,
  1031→        [("content-type", "text/event-stream")],
  1032→        sse,
  1033→    )
  1034→}

  1252→// ---------------------------------------------------------------------------
  1253→// POST /admin/reload — hot-swap the native inference engine at runtime
  1254→// ---------------------------------------------------------------------------
  1255→
  1256→/// Reload the in-process GGUF model. Accepts an optional `{ model }` body:
  1257→///   - `{ model: "path.gguf" }` — load that file (also recorded via the
  1258→///     `AETHER_MODEL` env var so subsequent auto-detect loads keep using it).
  1259→///   - `{}` or omitted — reload the auto-detected/last model.
  1260→///
  1261→/// The OS calls this via `/api/alpha/reload-engine` when the AI (or a user)
  1262→/// emits a `reload_engine` mutation, letting the organism swap its own brain
  1263→/// without a process restart.
  1264→///
  1265→/// # Security & threading (audit fixes)
  1266→///
  1267→/// - The requested model path is validated to resolve INSIDE the engine's
  1268→///   `models/` discovery root ([`validate_model_path`]); `..` traversal,
  1269→///   absolute paths outside `models/`, and symlink escapes are rejected
  1270→///   with `400 { ok: false, error: … }`.
  1271→/// - The (heavy, blocking) model load runs in `tokio::task::spawn_blocking`
  1272→///   so it never stalls a tokio worker, and the `native_engine` swap-lock is
  1273→///   held across the load so no generation starts mid-swap and two reloads
  1274→///   cannot race.
  1275→/// - `std::env::set_var` is process-global state: calling it from an async
  1276→///   context races with concurrent `env::var` readers (UB on POSIX). It is
  1277→///   therefore executed inside the `spawn_blocking` closure, guarded by the
  1278→///   swap-lock, with `AETHER_TOKEN` snapshotted into `AppState` at startup —
  1279→///   see the inline THREAD-SAFETY note in the handler body.
  1280→pub async fn admin_reload(
  1281→    State(state): State<AppState>,
  1282→    body: Option<Json<serde_json::Value>>,
  1283→) -> impl IntoResponse {
  1284→    let model_arg = body
  1285→        .and_then(|Json(v)| v.get("model").and_then(|m| m.as_str()).map(String::from));
  1286→
  1287→    // --- Path validation (security): the requested model MUST resolve
  1288→    // inside the engine's models/ discovery root. Without a body the
  1289→    // reload auto-detects (unchanged behaviour).
  1290→    let validated = match model_arg.as_deref() {
  1291→        Some(m) => match validate_model_path(m) {
  1292→            Ok(p) => Some(p),
  1293→            Err(reason) => {
  1294→                eprintln!("[aether-engine] reload rejected: {reason}");
  1295→                return (
  1296→                    axum::http::StatusCode::BAD_REQUEST,
  1297→                    Json(json!({ "ok": false, "error": reason })),
  1298→                )
  1299→                    .into_response();
  1300→            }
  1301→        },
  1302→        None => None,
  1303→    };
  1304→
  1305→    // Serialize the swap: hold the swap-lock across the (expensive, blocking)
  1306→    // load. In-flight generations holding the old inner Arc keep running;
  1307→    // new ones simply wait for the swap to finish.
  1308→    let mut guard = state.native_engine.lock().await;
  1309→
  1310→    // --- Blocking work off the async runtime: loading a GGUF is a heavy
  1311→    // synchronous operation (mmap + tensor init); run it on a dedicated
  1312→    // blocking thread so tokio workers stay free.
  1313→    let model_for_env = validated;
  1314→    let load_result = tokio::task::spawn_blocking(move || {
  1315→        if let Some(path) = model_for_env {
  1316→            // THREAD-SAFETY: this is the ONLY runtime environment mutation
  1317→            // in the process, and it is safe here because
  1318→            //   (a) `AETHER_TOKEN` is snapshotted into `AppState` at startup
  1319→            //       (no request-path `env::var` reads remain), and
  1320→            //   (b) reloads are serialized by the `native_engine` swap-lock
  1321→            //       held by this handler, so no other thread can run this
  1322→            //       block (or `find_model_path`'s `AETHER_MODEL` read)
  1323→            //       concurrently.
  1324→            // Never call `std::env::set_var` from an async context — it is
  1325→            // process-global state and concurrent get/set is UB on POSIX.
  1326→            std::env::set_var("AETHER_MODEL", path.as_os_str());
  1327→        }
  1328→        crate::engine::NativeEngine::load(2048)
  1329→    })
  1330→    .await;
  1331→
  1332→    match load_result {
  1333→        Ok(Ok(eng)) => {
  1334→            // Swap the engine atomically. In-flight generations holding the
  1335→            // old inner Arc keep running until they drop; new requests pick
  1336→            // up the new engine.
  1337→            *guard = Some(std::sync::Arc::new(tokio::sync::Mutex::new(eng)));
  1338→            let msg = match model_arg {
  1339→                Some(m) => format!("engine reloaded → {m}"),
  1340→                None => "engine reloaded (auto-detected model)".to_string(),
  1341→            };
  1342→            eprintln!("[aether-engine] {msg}");
  1343→            Json(json!({ "ok": true, "message": msg })).into_response()
  1344→        }
  1345→        Ok(Err(e)) => {
  1346→            eprintln!("[aether-engine] reload failed: {e}");
  1347→            Json(json!({ "ok": false, "error": e })).into_response()
  1348→        }
  1349→        Err(e) => {
  1350→            // spawn_blocking join error (the load panicked) — same JSON shape.
  1351→            eprintln!("[aether-engine] reload task failed: {e}");
  1352→            Json(json!({
  1353→                "ok": false,
  1354→                "error": format!("engine reload task failed: {e}")
  1355→            }))
  1356→            .into_response()
  1357→        }
  1358→    }
  1359→}
  1360→
  1361→/// Resolve the `models/` discovery root — mirrors the candidate walk in
  1362→/// [`crate::engine::NativeEngine::find_model_path`] (cwd → exe dir →
  1363→/// `CARGO_MANIFEST_DIR` → `.`) so `/admin/reload` validates against the same
  1364→/// directory the loader auto-detects from. Falls back to a relative
  1365→/// `models/` (which then fails canonicalization with a clear error) when no
  1366→/// candidate exists yet.
  1367→fn models_root() -> std::path::PathBuf {
  1368→    let candidates = [
  1369→        std::env::current_dir().ok(),
  1370→        std::env::current_exe()
  1371→            .ok()
  1372→            .and_then(|p| p.parent().map(|p| p.to_path_buf())),
  1373→        Some(std::path::PathBuf::from(env!("CARGO_MANIFEST_DIR"))),
  1374→        Some(std::path::PathBuf::from(".")),
  1375→    ];
  1376→    for base in candidates.into_iter().flatten() {
  1377→        let dir = base.join("models");
  1378→        if dir.is_dir() {
  1379→            return dir;
  1380→        }
  1381→    }
  1382→    std::path::PathBuf::from("models")
  1383→}
  1384→
  1385→/// Validate a model path requested via `POST /admin/reload`.
  1386→///
  1387→/// The path must resolve INSIDE the engine's `models/` discovery root:
  1388→///
  1389→/// - absolute paths must already live under `models/`;
  1390→/// - relative paths are resolved against `models/` (a leading `models/`
  1391→///   component is tolerated too, resolved against the root's parent);
  1392→/// - `..` traversal and symlinks that escape the root are rejected — the
  1393→///   containment check runs on the CANONICALIZED path, so both are caught;
  1394→/// - the file must exist, be a regular file, and end in `.gguf`.
  1395→///
  1396→/// Returns the canonical path on success, or a human-readable reason for
  1397→/// the `400` response.
  1398→fn validate_model_path(requested: &str) -> Result<std::path::PathBuf, String> {
  1399→    if requested.trim().is_empty() {
  1400→        return Err("model path is empty".to_string());
  1401→    }
  1402→    let root = models_root();
  1403→    let canonical_root = root
  1404→        .canonicalize()
  1405→        .map_err(|e| format!("models/ directory not accessible: {e}"))?;
  1406→
  1407→    let requested_path = std::path::Path::new(requested);
  1408→    // Candidate resolutions, first valid match inside the root wins:
  1409→    //   absolute → as-is (must already be inside models/)
  1410→    //   relative → <models>/<path>            ("my-model.gguf")
  1411→    //   relative → <parent-of-models>/<path>  ("models/my-model.gguf")
  1412→    let mut candidates: Vec<std::path::PathBuf> = Vec::new();
  1413→    if requested_path.is_absolute() {
  1414→        candidates.push(requested_path.to_path_buf());
  1415→    } else {
  1416→        candidates.push(root.join(requested_path));
  1417→        if let Some(base) = canonical_root.parent() {
  1418→            candidates.push(base.join(requested_path));
  1419→        }
  1420→    }
  1421→
  1422→    for candidate in candidates {
  1423→        // `canonicalize` also resolves `..` and symlinks, so the containment
  1424→        // check below cannot be fooled by either.
  1425→        let Ok(canonical) = candidate.canonicalize() else {
  1426→            continue; // doesn't exist — try the next resolution shape
  1427→        };
  1428→        if !canonical.is_file() {
  1429→            return Err(format!("not a file: {requested}"));
  1430→        }
  1431→        if !canonical.starts_with(&canonical_root) {
  1432→            return Err(format!(
  1433→                "model path escapes the models/ directory: {requested}"
  1434→            ));
  1435→        }
  1436→        let is_gguf = canonical
  1437→            .extension()
  1438→            .and_then(|e| e.to_str())
  1439→            .map(|e| e.eq_ignore_ascii_case("gguf"))
  1440→            .unwrap_or(false);
  1441→        if !is_gguf {
  1442→            return Err(format!("model must be a .gguf file: {requested}"));
  1443→        }
  1444→        return Ok(canonical);
  1445→    }
  1446→
  1447→    Err(format!("model file not found: {requested}"))
  1448→}

  1481→/// Detailed statistics about the semantic memory graph.
  1482→///
  1483→/// Returns:
  1484→///
  1485→/// - `node_count` — total nodes.
  1486→/// - `edge_count` — total directed edges in the adjacency list (each

  1489→/// - `max_edges` — the largest adjacency list of any single node (capped at
  1490→///   `top_k = 5`).
  1491→/// - `density` — `edge_count / (N * (N-1))` for `N > 1`, else 0 (the
  1492→///   directed-edge density of the adjacency list).
  1493→/// - `top_connected_nodes` — the top 5 nodes by adjacency-list length, each
  1494→///   as `{ id, text, edge_count }`.
  1495→pub async fn graph_stats(State(state): State<AppState>) -> impl IntoResponse {
  1496→    let g = state.graph.lock().await;
  1497→    let node_count = g.len();
  1498→    let edge_count = g.edge_count();
  1499→    let max_edges = g
  1500→        .adjacency
  1501→        .values()
  1502→        .map(|v| v.len())
  1503→        .max()
  1504→        .unwrap_or(0);
  1505→    let avg_edges_per_node = if node_count > 0 {
  1506→        edge_count as f64 / node_count as f64
  1507→    } else {
  1508→        0.0
  1509→    };
  1510→    let density = if node_count > 1 {
  1511→        edge_count as f64 / (node_count as f64 * (node_count as f64 - 1.0))
  1512→    } else {
  1513→        0.0
  1514→    };
  1515→
  1516→    // Top connected nodes: sort adjacency entries by length (desc), take 5.
  1517→    // We clone the (id, edge_count) pairs out of the adjacency map so we can
  1518→    // drop the borrow before looking up node texts (avoiding a double-borrow
  1519→    // of `g.adjacency` and `g.nodes`).
  1520→    let mut node_edge_counts: Vec<(String, usize)> = g
  1521→        .adjacency
  1522→        .iter()
  1523→        .map(|(id, nbrs)| (id.clone(), nbrs.len()))
  1524→        .collect();
  1525→    node_edge_counts.sort_by(|a, b| b.1.cmp(&a.1));
  1526→
  1527→    let top_connected: Vec<serde_json::Value> = node_edge_counts
  1528→        .into_iter()
  1529→        .take(5)
  1530→        .filter_map(|(id, ec)| {
  1531→            g.nodes.get(&id).map(|n| {
  1532→                json!({
  1533→                    "id": id,
  1534→                    "text": n.text.clone(),
  1535→                    "edge_count": ec,
  1536→                })
  1537→            })
  1538→        })
  1539→        .collect();
  1540→
  1541→    Json(json!({
  1542→        "node_count": node_count,
  1543→        "edge_count": edge_count,
  1544→        "avg_edges_per_node": avg_edges_per_node,
  1545→        "max_edges": max_edges,
  1546→        "density": density,
  1547→        "top_connected_nodes": top_connected,
  1548→    }))
  1549→}

  1559→// ---------------------------------------------------------------------------
  1560→// POST /v1/agent/run — run the autonomous agent loop
  1561→// ---------------------------------------------------------------------------
  1562→
  1563→/// Run the autonomous agent loop for a single goal.
  1564→///
  1565→/// Accepts a JSON body of the shape:
  1566→///
  1567→/// ```json
  1568→/// {
  1569→///   "goal": "set up a Python dev environment and write a hello-world script",
  1570→///   "context": { "windows": [...], "memory": [...], "plan": {...} },
  1571→///   "max_iterations": 20
  1572→/// }
  1573→/// ```
  1574→///
  1575→/// - `goal` (required) — natural-language goal the agent should achieve.
  1576→/// - `context` (optional, default `{}`) — current OS state, pretty-printed
  1577→///   into the system prompt so the LLM perceives open windows, memory,
  1578→///   active plan, etc.
  1579→/// - `max_iterations` (optional, default `20`, clamped to `[1, 50]`) —
  1580→///   hard cap on think iterations.
  1581→///
  1582→/// Returns:
  1583→///
  1584→/// ```json
  1585→/// {
  1586→///   "ok": true,
  1587→///   "result": "<final assistant text>",
  1588→///   "iterations": 3,
  1589→///   "completed": true,
  1590→///   "tool_calls": [
  1591→///     { "name": "file_read", "params": { "path": "/etc/hostname" } },
  1592→///     ...
  1593→///   ]
  1594→/// }
  1595→/// ```
  1596→///
  1597→/// The `tool_calls` array is the full transcript of every tool call the
  1598→/// LLM made across all iterations, in execution order. The Next.js layer
  1599→/// uses this transcript to perform the real OS side-effects (file IO,
  1600→/// PTY exec, window manager mutations, …) — the Aether Engine's tool
  1601→/// executors return placeholders by design (see [`crate::tools`]).
  1602→///
  1603→/// On a missing `goal` field the handler returns
  1604→/// `{ ok: false, error: "missing 'goal' field" }` without starting the
  1605→/// loop.
  1606→pub async fn agent_run(
  1607→    State(state): State<AppState>,
  1608→    Json(body): Json<serde_json::Value>,
  1609→) -> impl IntoResponse {
  1610→    let goal = body
  1611→        .get("goal")
  1612→        .and_then(|g| g.as_str())
  1613→        .unwrap_or("")
  1614→        .to_string();
  1615→    let context = body.get("context").cloned().unwrap_or(json!({}));
  1616→    let max_iterations = body
  1617→        .get("max_iterations")
  1618→        .and_then(|m| m.as_u64())
  1619→        .map(|n| n as usize)
  1620→        .unwrap_or(crate::agent::DEFAULT_MAX_ITERATIONS);
  1621→
  1622→    if goal.is_empty() {
  1623→        return Json(json!({
  1624→            "ok": false,
  1625→            "error": "missing 'goal' field",
  1626→        }))
  1627→        .into_response();
  1628→    }
  1629→
  1630→    let result = crate::agent::run_agent_loop(state, &goal, &context, max_iterations).await;
  1631→
  1632→    Json(json!({
  1633→        "ok": true,
  1634→        "result": result.result,
  1635→        "iterations": result.iterations,
  1636→        "completed": result.completed,
  1637→        "tool_calls": result.tool_calls.iter().map(|c| json!({
  1638→            "name": c.name,
  1639→            "params": c.params,
  1640→        })).collect::<Vec<_>>(),
  1641→    }))
  1642→    .into_response()
  1643→}

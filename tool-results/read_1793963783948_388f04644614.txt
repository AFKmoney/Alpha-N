     1→     1→/**
     2→     2→ * autonomous-loop.tsx — the real AI driver. Captures a screenshot of the
     3→     3→ * desktop, sends it + full OS state to /api/alpha/think, then applies the
     4→     4→ * returned mutations one-by-one. Handles web_search, read_file, write_file,
     5→     5→ * debate, execute_code, compile, and rollback tools. Runs on a 22s cadence
     6→     6→ * in "active" mode and reacts to user chat / unhandled events in both modes.
     7→     7→ */
     8→     8→"use client";
     9→     9→
    10→    10→import { useCallback, useEffect, useRef } from "react";
    11→    11→import { useEvolution } from "@/lib/alpha/evolution-store";
    12→    12→import { useOS } from "@/lib/alpha/os-store";
    13→    13→import type { AppKind } from "@/lib/alpha/os-types";
    14→    14→import { captureScreenshot, think, webSearch, readFile, writeFile, runDebate, executeCode, runCompile, auditAction } from "@/lib/alpha/ai-client";
    15→    15→import { describeMutation, type Mutation, type BeforeAfter, type WebSearchResult, type CodeExecResult, type CompileResult, type DebateResult, type MutationRewardEntry, type EpisodeEntry } from "@/lib/alpha/mutations";
    16→    16→import { getPolicy, authorize, isConsequential, type AutonomyLevel } from "@/lib/alpha/autonomy-policy";
    17→    17→import { computeReward } from "@/lib/alpha/reward-model";
    18→    18→import { buildLesson, derivePersistedLesson } from "@/lib/alpha/learning-engine";
    19→    19→
    20→    20→const CYCLE_MS = 22000; // autonomous cycle cadence — responsive real-time control
    21→    21→const MUTATION_STEP_MS = 320;
    22→    22→
    23→    23→export function AutonomousLoop({ workspaceRef }: { workspaceRef: React.RefObject<HTMLElement | null> }) {
    24→    24→  const store = useEvolution;
    25→    25→  const osStore = useOS;
    26→    26→  const runningRef = useRef(false);
    27→    27→  const consecutiveErrorsRef = useRef(0); // for exponential backoff on rate limits
    28→    28→
    29→    29→  const runCycle = useCallback(
    30→    30→    async (userMessage?: string) => {
    31→    31→      if (runningRef.current) return;
    32→    32→      runningRef.current = true;
    33→    33→
    34→    34→      const st = store.getState();
    35→    35→      const os = osStore.getState();
    36→    36→      const { setAiBusy, applyMutation, setBeforeAfter } = st;
    37→    37→
    38→    38→      setAiBusy(true, "Observing my own desktop…");
    39→    39→
    40→    40→      // ---- SNAPSHOT before mutations (for rollback) ----
    41→    41→      const curState = store.getState();
    42→    42→      const curOs = osStore.getState();
    43→    43→      const snapshot = os.takeSnapshot(`pre-cycle gen ${curState.generation}`, {
    44→    44→        windows: curOs.windows,
    45→    45→        codeLines: curState.codeLines,
    46→    46→        agents: curState.agents,
    47→    47→        metrics: curState.metrics,
    48→    48→        version: curState.version,
    49→    49→        generation: curState.generation,
    50→    50→      });
    51→    51→
    52→    52→      // 1. BEFORE screenshot — the AI's persistent visual of its own body
    53→    53→      const before = await captureScreenshot(workspaceRef.current, "before");
    54→    54→
    55→    55→      // 2. FULL OS state for the LLM (1M token consciousness)
    56→    56→      const s = store.getState();
    57→    57→      const osS = osStore.getState();
    58→    58→      const fullCode = s.codeLines
    59→    59→        .map((l) => {
    60→    60→          const text = l.tokens.map((t) => t.text).join("");
    61→    61→          return `${String(l.no).padStart(2, "0")}│ ${text}`;
    62→    62→        })
    63→    63→        .join("\n");
    64→    64→
    65→    65→      setAiBusy(true, "Consulting the cognitive council…");
    66→    66→
    67→    67→      // 3. think — feed the FULL OS context: all code, all logs, all mutations,
    68→    68→      //    full chat history, all windows with positions, Akasha memory + intentions,
    69→    69→      //    protected files, search results. GLM 5.2 has 1M tokens — use them.
    70→    70→      let result;
    71→    71→      try {
    72→    72→        result = await think({
    73→    73→          screenshot: before,
    74→    74→          state: {
    75→    75→            generation: s.generation,
    76→    76→            version: s.version,
    77→    77→            aiState: s.aiState,
    78→    78→            metrics: s.metrics,
    79→    79→            autonomyMode: s.autonomyMode,
    80→    80→            fullCode,
    81→    81→            agents: s.agents.map((a) => ({
    82→    82→              role: a.role,
    83→    83→              status: a.status,
    84→    84→              thought: a.thought,
    85→    85→              load: a.load,
    86→    86→            })),
    87→    87→            allLogs: s.logs.map((l) => `[${l.level}] ${l.agent}: ${l.message}`),
    88→    88→            allMutations: s.mutationStream.map((m) => m.description),
    89→    89→            windows: osS.windows.map((w) => ({
    90→    90→              id: w.id,
    91→    91→              kind: w.kind,
    92→    92→              title: w.title,
    93→    93→              x: Math.round(w.x),
    94→    94→              y: Math.round(w.y),
    95→    95→              w: Math.round(w.w),
    96→    96→              h: Math.round(w.h),
    97→    97→              desktop: w.desktop,
    98→    98→            })),
    99→    99→            violations: osS.violationAttempts.map((v) => ({ path: v.path, reason: v.reason })),
   100→   100→            rollbacks: osS.rollbackEvents.length,
   101→   101→            searchResults: s.searchResults.slice(0, 3).map((sr) => ({
   102→   102→              query: sr.query,
   103→   103→              top: sr.results.slice(0, 4).map((r) => `${r.title} — ${r.snippet.slice(0, 120)} (${r.host})`),
   104→   104→            })),
   105→   105→            akashaMemory: s.akashaMemory.map((m) => ({ text: m.text, kind: m.kind })),
   106→   106→            akashaIntentions: s.akashaIntentions.map((i) => ({ text: i.text, priority: i.priority, resolved: i.resolved })),
   107→   107→            dynamicPrompt: s.dynamicPrompt,
   108→   108→            plans: s.plans.map((p) => ({
   109→   109→              id: p.id,
   110→   110→              goal: p.goal,
   111→   111→              rationale: p.rationale,
   112→   112→              status: p.status,
   113→   113→              steps: p.steps,
   114→   114→              progress: `${p.steps.filter((st) => st.done).length}/${p.steps.length}`,
   115→   115→            })),
   116→   116→            goals: s.goals.map((g) => ({ text: g.text, level: g.level })),
   117→   117→            fileReads: s.fileReads.slice(0, 4).map((f) => ({ path: f.path, content: f.content.slice(0, 2000) })),
   118→   118→            debateResults: s.debateResults.slice(0, 2).map((d) => ({
   119→   119→              proposal: d.proposal,
   120→   120→              consensus: d.consensus,
   121→   121→              opinions: d.opinions.map((o) => `${o.agent}: ${o.verdict} — ${o.opinion.slice(0, 100)}`),
   122→   122→            })),
   123→   123→            execResults: s.execResults.slice(0, 2).map((e) => ({
   124→   124→              language: e.language,
   125→   125→              ok: e.ok,
   126→   126→              stdout: e.stdout.slice(0, 500),
   127→   127→              stderr: e.stderr.slice(0, 300),
   128→   128→            })),
   129→   129→            compileResults: s.compileResults.slice(0, 2).map((c) => ({
   130→   130→              check: c.check,
   131→   131→              ok: c.ok,
   132→   132→              tscOutput: (c.tscOutput ?? "").slice(0, 400),
   133→   133→              eslintOutput: (c.eslintOutput ?? "").slice(0, 400),
   134→   134→            })),
   135→   135→            rewardModel: s.rewardModel.slice(0, 10).map((r) => ({
   136→   136→              kind: r.kind,
   137→   137→              delta: r.delta.toFixed(3),
   138→   138→              helpful: r.helpful,
   139→   139→            })),
   140→   140→            events: s.eventQueue.filter((e) => !e.handled).slice(0, 5).map((e) => ({ type: e.type, content: e.content.slice(0, 300) })),
   141→   141→            protectedFiles: osS.protectedFiles.map((f) => ({ path: f.path, reason: f.reason, critical: f.critical })),
   142→   142→            desktops: 4,
   143→   143→            activeDesktop: osS.activeDesktop,
   144→   144→            layoutMode: osS.layoutMode,
   145→   145→            // ---- Phase 1/3/4: episodic memory + metrics + constraints ----
   146→   146→            episodeLog: s.episodeLog.slice(-20).map((e) => ({
   147→   147→              action: e.action,
   148→   148→              description: e.description,
   149→   149→              result: e.result,
   150→   150→              reward: e.reward,
   151→   151→              cycle: e.cycle,
   152→   152→            })),
   153→   153→            realMetrics: s.realMetrics,
   154→   154→            constraints: s.constraints.map((c) => ({ text: c.text, scope: c.scope })),
   155→   155→            // Send accumulated lessons from PREVIOUS cycles so the AI reads
   156→   156→            // its own feedback before deciding what to do this cycle.
   157→   157→            persistedLessons: s.persistedLessons,
   158→   158→          },
   159→   159→          userMessage: userMessage ?? null,
   160→   160→          history: s.chat.map((m) => ({ role: m.role, content: m.content })),
   161→   161→        });
   162→   162→      } catch (err) {
   163→   163→        const msg = err instanceof Error ? err.message : "unknown";
   164→   164→        applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `Cognitive call failed: ${msg.slice(0, 70)}` });
   165→   165→        setAiBusy(false);
   166→   166→        runningRef.current = false;
   167→   167→        return;
   168→   168→      }
   169→   169→
   170→   170→      // 4. Handle rate-limiting / errors gracefully
   171→   171→      if (result.rateLimited || (result.error && result.mutations.length === 0)) {
   172→   172→        // The cognitive API is rate-limited or errored. Log ONCE, don't speak
   173→   173→        // (avoid chat noise), and set a backoff flag so the next cycle waits longer.
   174→   174→        const logLevel = result.rateLimited ? "critique" : "critique";
   175→   175→        const logMsg = result.rateLimited
   176→   176→          ? "Cognitive API rate-limited (429). Backing off — will retry with longer delay."
   177→   177→          : `Cognitive error: ${(result.error ?? "unknown").slice(0, 70)}`;
   178→   178→        // Only log if the last log isn't already the same (avoid log spam)
   179→   179→        const lastLog = store.getState().logs[0];
   180→   180→        if (!lastLog || !lastLog.message.includes("rate-limited") && !lastLog.message.includes("Cognitive error")) {
   181→   181→          applyMutation({ type: "add_log", level: logLevel, agent: "nucleus", message: logMsg });
   182→   182→        }
   183→   183→        // Set the backoff flag — the cycle useEffect will use a longer delay
   184→   184→        consecutiveErrorsRef.current = Math.min(consecutiveErrorsRef.current + 1, 5);
   185→   185→        setAiBusy(false);
   186→   186→        runningRef.current = false;
   187→   187→        return;
   188→   188→      }
   189→   189→
   190→   190→      // Success — reset error counter
   191→   191→      consecutiveErrorsRef.current = 0;
   192→   192→
   193→   193→      // 5. apply mutations one-by-one
   194→   194→      setAiBusy(true, result.reasoning || "Applying mutations to my own source…");
   195→   195→
   196→   196→      const mutations: Mutation[] = Array.isArray(result.mutations) ? result.mutations : [];
   197→   197→      // Only auto-add a speak mutation if there are real mutations AND a message.
   198→   198→      // Don't speak error messages (those are handled above).
   199→   199→      if (result.message && mutations.length > 0) {
   200→   200→        const hasSpeak = mutations.some((m) => m.type === "speak");
   201→   201→        if (!hasSpeak) {
   202→   202→          mutations.push({ type: "speak", message: result.message, reasoning: result.reasoning });
   203→   203→        }
   204→   204→      }
   205→   205→
   206→   206→      // Track if any mutation was rejected/blocked
   207→   207→      let hadError = false;
   208→   208→      const streamBefore = store.getState().mutationStream.length;
   209→   209→
   210→   210→      // Resolve the active autonomy policy once per cycle. Consequential
   211→   211→      // mutations (file writes, exec, self-prompt, delete) are gated by it.
   212→   212→      const currentLevel: AutonomyLevel = store.getState().autonomyLevel;
   213→   213→      const policy = getPolicy(currentLevel);
   214→   214→      // Simple client-side rate limiter for consequential actions.
   215→   215→      const actionTimestamps: number[] = [];
   216→   216→      const rateLimitMs = 60_000;
   217→   217→
   218→   218→      // ---- OBJECTIVE REWARD TRACKING ----
   219→   219→      // Per-mutation outcome sets, populated as the loop runs. These feed
   220→   220→      // the objective reward model (verifiable signals, not self-report).
   221→   221→      const toolOk = new Set<string>(); // mutations whose tool call succeeded
   222→   222→      const toolError = new Set<string>(); // mutations whose tool errored
   223→   223→      const blocked = new Set<string>(); // mutations denied by policy/security
   224→   224→
   225→   225→      for (let mi = 0; mi < mutations.length; mi++) {
   226→   226→        const m = mutations[mi];
   227→   227→        const mType = (m as { type?: string }).type ?? "unknown";
   228→   228→
   229→   229→        // ---- AUTONOMY POLICY GATE ----
   230→   230→        // Every mutation is authorised against the active policy. Denied
   231→   231→        // mutations are logged and skipped — the AI is told why so it can
   232→   232→        // adapt (e.g. switch approach or ask the user to raise the level).
   233→   233→        // For self-escalation mutations we pass the requested target so the
   234→   234→        // policy can block a yolo power-grab made from a lower level.
   235→   235→        const authArgs =
   236→   236→          mType === "set_autonomy_level" && m && typeof m === "object" && "level" in m
   237→   237→            ? { target: String((m as { level: string }).level) }
   238→   238→            : undefined;
   239→   239→        const verdict = authorize(mType, policy, authArgs);
   240→   240→        if (!verdict.allowed) {
   241→   241→          blocked.add(String(mi));
   242→   242→          applyMutation({
   243→   243→            type: "add_log",
   244→   244→            level: "critique",
   245→   245→            agent: "nucleus",
   246→   246→            message: `⊘ BLOCKED by autonomy policy [${policy.level}]: ${verdict.reason}`,
   247→   247→          });
   248→   248→          // Record to the mutation stream as a denial so the AI sees it.
   249→   249→          store.getState().applyMutation({
   250→   250→            type: "add_log",
   251→   251→            level: "critique",
   252→   252→            agent: "critic",
   253→   253→            message: `Denied "${mType}": ${verdict.reason}`,
   254→   254→          });
   255→   255→          continue;
   256→   256→        }
   257→   257→
   258→   258→        // ---- RATE LIMIT (consequential actions only) ----
   259→   259→        if (isConsequential(mType) && policy.capabilities.actionsPerMinute > 0) {
   260→   260→          const now = Date.now();
   261→   261→          // Drop timestamps older than the window.
   262→   262→          while (actionTimestamps.length && now - actionTimestamps[0] > rateLimitMs) {
   263→   263→            actionTimestamps.shift();
   264→   264→          }
   265→   265→          if (actionTimestamps.length >= policy.capabilities.actionsPerMinute) {
   266→   266→            blocked.add(String(mi));
   267→   267→            applyMutation({
   268→   268→              type: "add_log",
   269→   269→              level: "critique",
   270→   270→              agent: "critic",
   271→   271→              message: `⊘ Rate limit hit (${policy.capabilities.actionsPerMinute}/min under ${policy.level}). Deferring "${mType}".`,
   272→   272→            });
   273→   273→            continue;
   274→   274→          }
   275→   275→          actionTimestamps.push(now);
   276→   276→        }
   277→   277→
   278→   278→        applyMutation(m as Mutation);
   279→   279→        await sleep(MUTATION_STEP_MS);
   280→   280→        // check if the latest mutation stream entry is a violation
   281→   281→        const stream = store.getState().mutationStream;
   282→   282→        if (stream.length > streamBefore) {
   283→   283→          const latest = stream[0];
   284→   284→          if (latest.kind === "violation") {
   285→   285→            hadError = true;
   286→   286→          }
   287→   287→        }
   288→   288→        // ---- WEB SEARCH tool: when the AI emits a web_search mutation,
   289→   289→        // perform the search and store results for the next cycle ----
   290→   290→        if (m.type === "web_search" && m.query) {
   291→   291→          setAiBusy(true, `Searching the web: "${m.query.slice(0, 50)}"…`);
   292→   292→          try {
   293→   293→            const searchRes = await webSearch(m.query);
   294→   294→            const wsr: WebSearchResult = {
   295→   295→              query: m.query,
   296→   296→              time: Date.now(),
   297→   297→              results: searchRes.results,
   298→   298→            };
   299→   299→            store.getState().addSearchResults(wsr);
   300→   300→            // also open a browser window showing the first result so the user sees the research
   301→   301→            if (searchRes.results[0]?.url) {
   302→   302→              store.getState().applyMutation({
   303→   303→                type: "create_app",
   304→   304→                appType: "browser",
   305→   305→                title: `🔍 ${m.query.slice(0, 30)}`,
   306→   306→                url: searchRes.results[0].url,
   307→   307→              });
   308→   308→            }
   309→   309→          } catch (err) {
   310→   310→            const msg = err instanceof Error ? err.message : "search failed";
   311→   311→            applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `Web search failed: ${msg.slice(0, 60)}` });
   312→   312→          }
   313→   313→        }
   314→   314→        // ---- FILE READ tool: when the AI emits read_file, read the real file ----
   315→   315→        if (m.type === "read_file" && m.path) {
   316→   316→          setAiBusy(true, `Reading file: ${m.path}…`);
   317→   317→          try {
   318→   318→            const res = await readFile(m.path);
   319→   319→            if (res.type === "file" && res.content !== undefined) {
   320→   320→              store.getState().addFileRead({
   321→   321→                path: m.path,
   322→   322→                content: res.content,
   323→   323→                time: Date.now(),
   324→   324→              });
   325→   325→            } else if (res.type === "dir" && res.entries) {
   326→   326→              store.getState().addFileRead({
   327→   327→                path: m.path,
   328→   328→                content: `Directory listing:\n${res.entries.map((e) => `${e.isDir ? "[DIR] " : "      "}${e.name}`).join("\n")}`,
   329→   329→                time: Date.now(),
   330→   330→              });
   331→   331→            } else if (res.error) {
   332→   332→              applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `File read error: ${res.error.slice(0, 60)}` });
   333→   333→            }
   334→   334→          } catch (err) {
   335→   335→            const msg = err instanceof Error ? err.message : "read failed";
   336→   336→            applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `File read failed: ${msg.slice(0, 60)}` });
   337→   337→          }
   338→   338→        }
   339→   339→        // ---- FILE WRITE tool: when the AI emits write_file, write the real file ----
   340→   340→        if (m.type === "write_file" && m.path) {
   341→   341→          setAiBusy(true, `Writing file: ${m.path}…`);
   342→   342→          try {
   343→   343→            const res = await writeFile(m.path, m.content);
   344→   344→            if (res.error) {
   345→   345→              toolError.add(String(mi));
   346→   346→              blocked.add(String(mi));
   347→   347→              applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `File write BLOCKED: ${res.error.slice(0, 60)}` });
   348→   348→              osStore.getState().recordViolation(m.path, `AI tried to write protected file: ${m.path}`);
   349→   349→            } else {
   350→   350→              toolOk.add(String(mi));
   351→   351→              applyMutation({ type: "add_log", level: "deploy", agent: "developer", message: `Wrote ${m.content.length} bytes to ${m.path}`, });
   352→   352→            }
   353→   353→          } catch (err) {
   354→   354→            toolError.add(String(mi));
   355→   355→            const msg = err instanceof Error ? err.message : "write failed";
   356→   356→            applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `File write failed: ${msg.slice(0, 60)}` });
   357→   357→          }
   358→   358→        }
   359→   359→        // ---- LAYER A: AGENT DEBATE ----
   360→   360→        if (m.type === "debate" && m.proposal) {
   361→   361→          setAiBusy(true, `Convening the council to debate…`);
   362→   362→          try {
   363→   363→            const debateRes = await runDebate(
   364→   364→              m.proposal,
   365→   365→              `Generation ${store.getState().generation}, coherence ${store.getState().metrics.coherence.toFixed(2)}, entropy ${store.getState().metrics.entropy.toFixed(2)}`,
   366→   366→              store.getState().mutationStream.slice(0, 5).map((mut) => mut.description)
   367→   367→            );
   368→   368→            const dr: DebateResult = {
   369→   369→              proposal: m.proposal,
   370→   370→              opinions: debateRes.opinions.map((o) => ({
   371→   371→                agent: o.agent,
   372→   372→                opinion: o.opinion,
   373→   373→                verdict: o.verdict as "PROCEED" | "REVISE" | "REJECT",
   374→   374→              })),
   375→   375→              consensus: debateRes.consensus as "PROCEED" | "REVISE" | "REJECT",
   376→   376→              tally: debateRes.tally,
   377→   377→              time: Date.now(),
   378→   378→            };
   379→   379→            store.getState().addDebateResult(dr);
   380→   380→            // If the council says REJECT, push a log so the AI knows
   381→   381→            if (dr.consensus === "REJECT") {
   382→   382→              applyMutation({ type: "add_log", level: "critique", agent: "critic", message: `Council REJECTED: "${m.proposal.slice(0, 50)}". Do not proceed.` });
   383→   383→            }
   384→   384→          } catch (err) {
   385→   385→            const msg = err instanceof Error ? err.message : "debate failed";
   386→   386→            applyMutation({ type: "add_log", level: "critique", agent: "nucleus", message: `Debate failed: ${msg.slice(0, 60)}` });
   387→   387→          }
   388→   388→        }
   389→   389→        // ---- LAYER C: SANDBOXED CODE EXECUTION ----
   390→   390→        if (m.type === "execute_code" && m.code) {
   391→   391→          setAiBusy(true, `Executing ${m.language} code…`);

   766→   766→          level: "heal",
   767→   767→          agent: "nucleus",
   768→   768→          message: `↺ ROLLBACK: ${reason} Restored to ${snapshot.label}.`,
   769→   769→        });
   770→   770→      }
   771→   771→
   772→   772→      // 5. AFTER screenshot
   773→   773→      await sleep(300);
   774→   774→      const after = await captureScreenshot(workspaceRef.current, "after");
   775→   775→
   776→   776→      // 6. before/after viewer
   777→   777→      if (before && after) {
   778→   778→        const cur = store.getState();
   779→   779→        const ba: BeforeAfter = {
   780→   780→          id: `ba-${Date.now()}`,
   781→   781→          label: `Generation ${cur.generation} · v${cur.version}`,
   782→   782→          before,

   784→   784→          time: Date.now(),
   785→   785→          summary: result.message || result.reasoning || "Self-improvement applied.",
   786→   786→        };
   787→   787→        setBeforeAfter(ba);
   788→   788→      }
   789→   789→
   790→   790→      setAiBusy(false);
   791→   791→      runningRef.current = false;
   792→   792→    },
   793→   793→    [store, osStore, workspaceRef]
   794→   794→  );
   795→   795→
   796→   796→  const autonomyMode = useEvolution((s) => s.autonomyMode);
   797→   797→  const aiBusy = useEvolution((s) => s.aiBusy);
   798→   798→  const activeEvolution = useEvolution((s) => s.activeEvolution);
   799→   799→  const forceCycle = useEvolution((s) => s.forceCycle);
   800→   800→  const chat = useEvolution((s) => s.chat);
   801→   801→  const hydrateFromDb = useEvolution((s) => s.hydrateFromDb);
   802→   802→  const loadEpisodesFromDb = useEvolution((s) => s.loadEpisodesFromDb);
   803→   803→  const loadConstraints = useEvolution((s) => s.loadConstraints);
   804→   804→  const eventQueue = useEvolution((s) => s.eventQueue);
   805→   805→
   806→   806→  // Hydrate persistent cognition (Akasha, plans, goals, episodes, constraints) on mount.
   807→   807→  useEffect(() => {
   808→   808→    void hydrateFromDb();
   809→   809→    void loadEpisodesFromDb();
   810→   810→    void loadConstraints();
   811→   811→  }, [hydrateFromDb, loadEpisodesFromDb, loadConstraints]);
   812→   812→
   813→   813→  // ---- BOOT: auto-detect a working LLM provider ----
   814→   814→  // The default config is "cloud", but the cloud SDK only works when a
   815→   815→  // .z-ai-config is present. If it isn't (common on a fresh checkout), the
   816→   816→  // organism would boot into a dead state — think() throws and the AI never
   817→   817→  // replies. Probe both providers on mount and let the server switch to
   818→   818→  // whichever is reachable (often the local Aether Engine).
   819→   819→  useEffect(() => {
   820→   820→    let cancelled = false;
   821→   821→    fetch("/api/alpha/probe-providers", { method: "POST", cache: "no-store" })
   822→   822→      .then((r) => (r.ok ? r.json() : null))
   823→   823→      .then((data) => {
   824→   824→        if (cancelled || !data) return;
   825→   825→        // Surface the decision in the logs so the user can see which brain
   826→   826→        // the organism actually booted with.
   827→   827→        if (data.switched) {
   828→   828→          useEvolution.setState((s) => ({
   829→   829→            logs: [
   830→   830→              {
   831→   831→                id: `log-boot-${Date.now()}`,
   832→   832→                time: Date.now(),
   833→   833→                level: "evolve" as const,
   834→   834→                agent: "nucleus" as const,
   835→   835→                message: `↺ Auto-switched LLM provider to "${data.activeProvider}" (the other wasn't reachable).`,
   836→   836→              },
   837→   837→              ...s.logs,
   838→   838→            ].slice(0, 80),
   839→   839→          }));
   840→   840→        }
   841→   841→      })
   842→   842→      .catch(() => {
   843→   843→        /* probe is best-effort; fallback logic in callLLM still applies */
   844→   844→      });
   845→   845→    return () => {
   846→   846→      cancelled = true;
   847→   847→    };
   848→   848→  }, []);
   849→   849→
   850→   850→  // Manual "evolve" button forces an immediate real AI cycle.
   851→   851→  useEffect(() => {
   852→   852→    if (!forceCycle) return;
   853→   853→    if (aiBusy) return;
   854→   854→    useEvolution.setState({ forceCycle: false });
   855→   855→    void runCycle();
   856→   856→  }, [forceCycle, aiBusy, runCycle]);
   857→   857→
   858→   858→  // ---- LAYER B: REACTIVE EVENTS — when unhandled events exist, fire an immediate cycle ----
   859→   859→  // Only in active mode — in standby, events are queued but don't trigger cycles.
   860→   860→  // The AI will process them when the user next asks it to do something.
   861→   861→  const unhandledCount = eventQueue.filter((e) => !e.handled).length;
   862→   862→  useEffect(() => {
   863→   863→    if (unhandledCount === 0) return;
   864→   864→    if (aiBusy) return;
   865→   865→    if (consecutiveErrorsRef.current > 0) return;
   866→   866→    if (autonomyMode !== "active") return; // standby = no reactive cycles
   867→   867→    const t = setTimeout(() => void runCycle(), 500);
   868→   868→    return () => clearTimeout(t);
   869→   869→  }, [unhandledCount, aiBusy, autonomyMode, runCycle]);
   870→   870→
   871→   871→  // ---- AUTONOMOUS CYCLE ----
   872→   872→  // In "standby" mode: the AI does NOT run autonomous cycles. It only acts when:
   873→   873→  //   1. The user sends a chat message (handled by the chat effect below)
   874→   874→  //   2. The user clicks "evolve" (handled by the forceCycle effect above)
   875→   875→  //   3. A reactive event fires (terminal error, compile error) — but ONLY if active
   876→   876→  //
   877→   877→  // In "active" mode: the AI runs autonomous cycles on a timer, works on tasks,
   878→   878→  //   uses the council, and can code freely on user-requested projects.
   879→   879→  //
   880→   880→  // EXCEPTION: Even in standby, if the AI discovers a critical self-upgrade via
   881→   881→  //   web search, it may test it in sandbox and apply it. This is handled by
   882→   882→  //   the think route's system prompt which instructs the AI accordingly.
   883→   883→  useEffect(() => {
   884→   884→    if (autonomyMode !== "active") return; // standby = no autonomous cycling
   885→   885→    if (aiBusy || activeEvolution) return;
   886→   886→    const errors = consecutiveErrorsRef.current;
   887→   887→    const delay = Math.min(errors > 0 ? CYCLE_MS * Math.pow(2, errors) : CYCLE_MS, 300000);
   888→   888→    const id = setTimeout(() => { void runCycle(); }, delay);
   889→   889→    return () => clearTimeout(id);
   890→   890→  }, [autonomyMode, aiBusy, activeEvolution, chat.length, runCycle]);
   891→   891→
   892→   892→  const lastChatLen = useRef(chat.length);
   893→   893→  useEffect(() => {
   894→   894→    if (chat.length > lastChatLen.current) {
   895→   895→      const last = chat[chat.length - 1];
   896→   896→      if (last?.role === "user") {
   897→   897→        void runCycle(last.content);
   898→   898→      }
   899→   899→    }
   900→   900→    lastChatLen.current = chat.length;
   901→   901→  }, [chat, runCycle]);
   902→   902→
   903→   903→  return null;
   904→   904→}

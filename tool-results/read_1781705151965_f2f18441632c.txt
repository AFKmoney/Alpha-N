     1→//! HTTP handlers for the Aether Engine v2.0.
     2→//!
     3→//! The chat_completions handler implements the full cognitive pipeline:
     4→//!   1. Action cache check (instant for repeated queries)
     5→//!   2. Graph retrieval (TF-IDF semantic search + edge expansion)
     6→//!   3. Context compression (40K→4K tokens preserving signal)
     7→//!   4. Complexity analysis (Simple / Moderate / Complex)
     8→//!   5. Cognitive decomposition (Complex queries are broken into sub-questions)
     9→//!   6. Sequential sub-question solving (each with fresh context + dependency injection)
    10→//!   7. Synthesis (combine sub-answers into final response)
    11→//!   8. Self-verification (check quality; retry once if failed)
    12→//!   9. Knowledge distillation (store successful decomposition patterns)
    13→//!  10. Speculative prefetch (warm cache for likely-next queries)

   124→pub async fn graph_add(
   125→    State(state): State<AppState>,
   126→    Json(req): Json<AddNodeRequest>,
   127→) -> impl IntoResponse {
   128→    let id = req.id.clone();
   129→    let text = req.text.clone();
   130→    let edges_created = {
   131→        let mut g = state.graph.lock().await;
   132→        g.add(req)
   133→    };
   134→    // Also fold into the Holographic Context Memory (HCM)
   135→    {
   136→        let mut hcm = state.hcm.lock().await;
   137→        let key = crate::hcm::hash_to_vector(&id, hcm.dim);
   138→        let value = crate::hcm::hash_to_vector(&text, hcm.dim);
   139→        hcm.fold(&key, &value);
   140→    }
   141→    state.stats.lock().await.hcm_pairs_folded += 1;
   142→    Json(json!({ "ok": true, "id": id, "edges_created": edges_created }))
   143→}

   183→// ---------------------------------------------------------------------------
   184→// POST /v1/chat/completions — the full cognitive pipeline (OpenAI-compatible)
   185→// ---------------------------------------------------------------------------
   186→pub async fn chat_completions(
   187→    State(state): State<AppState>,
   188→    Json(body): Json<serde_json::Value>,
   189→) -> impl IntoResponse {
   190→    let start = Instant::now();
   191→    state.stats.lock().await.requests += 1;
   192→
   193→    // Extract the last user message as the retrieval query.
   194→    let query = extract_user_query(&body);
   195→
   196→    // Vectorize the query against the graph's current vocabulary.
   197→    let qvec: SparseVec = {
   198→        let g = state.graph.lock().await;
   199→        g.vectorizer.vectorize(&query)
   200→    };
   201→
   202→    // ---- Stage 1: Action cache fast-path (similarity > 0.95) ----
   203→    {
   204→        let cache = state.cache.lock().await;
   205→        if let Some((resp, _sim)) = cache.get(&query, &qvec) {
   206→            drop(cache);
   207→            state.stats.lock().await.cache_hits += 1;
   208→            return Json(openai_completion(&resp, "aether-cache")).into_response();
   209→        }
   210→    }
   211→
   212→    // ---- Stage 2: Graph retrieval (TF-IDF + edge expansion) ----
   213→    let retrieved: Vec<ScoredNode> = {
   214→        let g = state.graph.lock().await;
   215→        g.retrieve(&query, 8)
   216→    };
   217→
   218→    // ---- Stage 3: Context compression (40K→4K preserving signal) ----
   219→    let context_block = {
   220→        let mut cache = state.cache.lock().await;
   221→        if let Some(ctx) = cache.get_retrieval(&query, &qvec) {
   222→            ctx
   223→        } else {
   224→            let ctx = compress::compress(&retrieved, &query, 6000);
   225→            cache.put_retrieval(&query, qvec.clone(), ctx.clone());
   226→            ctx
   227→        }
   228→    };
   229→
   230→    // ---- Stage 4: Complexity analysis ----
   231→    let complexity = decompose::analyze_complexity(&query);
   232→    let mut pipeline = PipelineState::new(&query);
   233→    pipeline.complexity = complexity.clone();
   234→
   235→    // ---- Stage 5-7: Cognitive decomposition + sequential solving + synthesis ----
   236→    let final_response = match complexity {
   237→        Complexity::Simple => {
   238→            // Simple query — one-shot through the backend
   239→            pipeline.stages_completed.push("one-shot".into());
   240→            call_backend(&state, &body, &context_block).await
   241→        }
   242→        Complexity::Moderate => {
   243→            // Moderate — two-step: think, then answer
   244→            state.stats.lock().await.decompositions += 1;
   245→            pipeline.stages_completed.push("two-step-think".into());
   246→            pipeline.stages_completed.push("two-step-answer".into());
   247→
   248→            // Step 1: Think about the problem
   249→            let think_body = inject_think_step(&body, &context_block, &query);
   250→            let think_response = call_backend(&state, &think_body, &context_block).await;
   251→            pipeline.total_backend_calls += 1;
   252→
   253→            // Step 2: Answer using the thinking result
   254→            let answer_body = inject_answer_step(&body, &think_response, &context_block);
   255→            call_backend(&state, &answer_body, &context_block).await
   256→        }
   257→        Complexity::Complex => {
   258→            // Complex — full decomposition pipeline
   259→            state.stats.lock().await.decompositions += 1;
   260→            pipeline.stages_completed.push("decompose".into());
   261→
   262→            // Check distillation store for a reusable decomposition pattern
   263→            let sub_questions = {
   264→                let distill = state.distillation.lock().await;
   265→                if let Some(subs) = distill.find(&query, &qvec) {
   266→                    state.stats.lock().await.distillation_hits += 1;
   267→                    pipeline.stages_completed.push("distillation-hit".into());
   268→                    subs
   269→                } else {
   270→                    // Fresh decomposition
   271→                    decompose::decompose(&query, &retrieved)
   272→                }
   273→            };
   274→
   275→            pipeline.sub_questions = sub_questions.clone();
   276→            pipeline.stages_completed.push("solve".into());
   277→
   278→            // Solve each sub-question sequentially, injecting dependency answers
   279→            let mut answers: HashMap<String, String> = HashMap::new();
   280→            for sub in &sub_questions {
   281→                let sub_prompt = decompose::build_sub_prompt(sub, &answers);
   282→                let sub_body = inject_sub_question(&body, &sub_prompt, &context_block);
   283→                let sub_answer = call_backend(&state, &sub_body, &context_block).await;
   284→                pipeline.total_backend_calls += 1;
   285→                answers.insert(sub.id.clone(), sub_answer);
   286→            }
   287→
   288→            // Synthesis: if there's a "synth" question, its answer is the final response
   289→            pipeline.stages_completed.push("synthesize".into());
   290→            if let Some(synth) = sub_questions.iter().find(|s| s.id == "synth") {
   291→                answers.get("synth").cloned().unwrap_or_default()
   292→            } else {
   293→                // No explicit synthesis — combine all answers
   294→                let combined: Vec<String> = sub_questions
   295→                    .iter()
   296→                    .filter_map(|s| answers.get(&s.id).cloned())
   297→                    .collect();
   298→                combined.join("\n\n")
   299→            }
   300→        }
   301→    };
   302→
   303→    pipeline.synthesis = Some(final_response.clone());
   304→
   305→    // ---- Stage 8: Asymmetric Tensor Dueling (ATD) verification ----
   306→    // Replaces the simple verify_response with a dual-graph collision:
   307→    // Graph A (likelihood) must overcome Graph B (entropy) for validation.
   308→    state.stats.lock().await.verifications += 1;
   309→    state.stats.lock().await.atd_verifications += 1;
   310→
   311→    let atd_config = crate::atd::ATDConfig::default();
   312→    let atd_result = crate::atd::verify(&final_response, &query, &atd_config);
   313→
   314→    let final_output = if atd_result.validated {
   315→        // ATD validated — response survived the likelihood-entropy collision
   316→        state.stats.lock().await.verifications_passed += 1;
   317→        state.stats.lock().await.atd_validated += 1;
   318→        pipeline.verification_passed = true;
   319→        pipeline.stages_completed.push("atd-validated".into());
   320→        final_response
   321→    } else {
   322→        // ATD rejected — retry based on the recommendation
   323→        state.stats.lock().await.atd_rejected += 1;
   324→        pipeline.stages_completed.push("atd-rejected-retry".into());
   325→
   326→        let retry_body = match atd_result.recommendation {
   327→            crate::atd::ATDRecommendation::FallBackToSimpleShot => {
   328→                // Model is confused — simplify
   329→                inject_simple_retry(&body, &context_block, &query)
   330→            }
   331→            _ => {
   332→                // Standard retry with more explicit prompt
   333→                inject_retry(&body, &context_block, &query)
   334→            }
   335→        };
   336→        let retry_response = call_backend(&state, &retry_body, &context_block).await;
   337→        pipeline.total_backend_calls += 1;
   338→
   339→        // Re-verify the retry with ATD
   340→        let retry_atd = crate::atd::verify(&retry_response, &query, &atd_config);
   341→        if retry_atd.validated {
   342→            state.stats.lock().await.verifications_passed += 1;
   343→            state.stats.lock().await.atd_validated += 1;
   344→            pipeline.verification_passed = true;
   345→            pipeline.stages_completed.push("atd-retry-validated".into());
   346→            retry_response
   347→        } else {
   348→            // Both attempts failed ATD — return the one with better collision delta
   349→            state.stats.lock().await.atd_rejected += 1;
   350→            pipeline.verification_passed = false;
   351→            pipeline.stages_completed.push("atd-fail-final".into());
   352→            if retry_atd.collision_delta > atd_result.collision_delta {
   353→                retry_response
   354→            } else {
   355→                final_response
   356→            }
   357→        }
   358→    };
   359→
   360→    pipeline.total_latency_ms = start.elapsed().as_millis() as u64;
   361→
   362→    // ---- Stage 9: Knowledge distillation (store successful patterns) ----
   363→    if matches!(complexity, Complexity::Complex) && pipeline.verification_passed {
   364→        let mut distill = state.distillation.lock().await;
   365→        distill.store(&query, qvec.clone(), pipeline.sub_questions.clone());
   366→    }
   367→
   368→    // ---- Cache the final response ----
   369→    state.cache.lock().await.put(&query, qvec.clone(), final_output.clone());
   370→
   371→    // ---- Stage 10: Speculative prefetch ----
   372→    prefetch(state.clone(), &retrieved).await;
   373→
   374→    Json(openai_completion(&final_output, "aether-pipeline")).into_response()
   375→}

   381→/// Forward a request to the GGUF backend, returning the assistant's text response.
   382→/// Falls back to a graph-only response if the backend is unreachable.
   383→async fn call_backend(
   384→    state: &AppState,
   385→    body: &serde_json::Value,
   386→    context_block: &str,
   387→) -> String {
   388→    let augmented = augment_messages(body, context_block);
   389→    let backend_url = format!("{}/chat/completions", state.backend.trim_end_matches('/'));
   390→    let client = state.client.clone();
   391→
   392→    match client.post(&backend_url).json(&augmented).send().await {
   393→        Ok(resp) if resp.status().is_success() => {
   394→            let txt = resp.text().await.unwrap_or_else(|_| "{}".to_string());
   395→            extract_assistant_content(&txt).unwrap_or_else(|| {
   396→                fallback_response(context_block, "empty backend response")
   397→            })
   398→        }
   399→        Ok(resp) => {
   400→            let status = resp.status();
   401→            let body_txt = resp.text().await.unwrap_or_default();
   402→            fallback_response(
   403→                context_block,
   404→                &format!("backend returned {status}: {}", body_txt.chars().take(200).collect::<String>()),
   405→            )
   406→        }
   407→        Err(e) => {
   408→            fallback_response(context_block, &format!("backend unreachable: {e}"))
   409→        }
   410→    }
   411→}

   417→/// Inject the Aether context block into the system message.
   418→fn augment_messages(body: &serde_json::Value, context_block: &str) -> serde_json::Value {
   419→    let mut out = body.clone();
   420→    let prompt = aether_prompt(context_block);
   421→    if let Some(messages) = out.get_mut("messages").and_then(|m| m.as_array_mut()) {
   422→        let mut injected = false;
   423→        for m in messages.iter_mut() {
   424→            if m.get("role").and_then(|r| r.as_str()) == Some("system") && !injected {
   425→                let existing = m
   426→                    .get("content")
   427→                    .and_then(|c| c.as_str())
   428→                    .unwrap_or("")
   429→                    .to_string();
   430→                let new_content = format!("{}\n\n{}", prompt, existing);
   431→                if let Some(obj) = m.as_object_mut() {
   432→                    obj.insert("content".to_string(), serde_json::Value::String(new_content));
   433→                }
   434→                injected = true;
   435→            }
   436→        }
   437→        if !injected {
   438→            messages.insert(0, json!({ "role": "system", "content": prompt }));
   439→        }
   440→    }
   441→    out
   442→}

   475→/// Inject a sub-question as the primary user message for decomposition.
   476→fn inject_sub_question(body: &serde_json::Value, sub_prompt: &str, context_block: &str) -> serde_json::Value {
   477→    let mut out = body.clone();
   478→    let sub_system = format!(
   479→        "{context_block}\n\n\
   480→         # AETHER COGNITIVE PIPELINE — SUB-QUESTION\n\
   481→         You are solving one step of a larger problem. Focus ONLY on this sub-question.\n\
   482→         Provide a clear, concise answer."
   483→    );
   484→    if let Some(messages) = out.get_mut("messages").and_then(|m| m.as_array_mut()) {
   485→        // Replace the system message
   486→        if let Some(sys) = messages.get_mut(0) {
   487→            if sys.get("role").and_then(|r| r.as_str()) == Some("system") {
   488→                if let Some(obj) = sys.as_object_mut() {
   489→                    obj.insert("content".to_string(), serde_json::Value::String(sub_system));
   490→                }
   491→            }
   492→        }
   493→        // Replace the last user message with the sub-question
   494→        if let Some(last) = messages.last_mut() {
   495→            if last.get("role").and_then(|r| r.as_str()) == Some("user") {
   496→                if let Some(obj) = last.as_object_mut() {
   497→                    obj.insert("content".to_string(), serde_json::Value::String(sub_prompt.to_string()));
   498→                }
   499→            }
   500→        }
   501→    }
   502→    out
   503→}

   520→/// Inject a simplified retry prompt for ATD FallBackToSimpleShot.
   521→/// Used when the model is deeply confused and needs a clean, simple instruction.
   522→fn inject_simple_retry(body: &serde_json::Value, context_block: &str, query: &str) -> serde_json::Value {
   523→    let mut out = body.clone();
   524→    let simple_prompt = format!(
   525→        "{context_block}\n\n\
   526→         # AETHER — SIMPLE MODE\n\
   527→         Answer this question directly and concisely: \"{query}\"\n\
   528→         Do not overthink. Provide a clear, short answer."
   529→    );
   530→    if let Some(messages) = out.get_mut("messages").and_then(|m| m.as_array_mut()) {
   531→        // Replace all messages with a clean, simple exchange
   532→        messages.clear();
   533→        messages.push(json!({ "role": "system", "content": simple_prompt }));
   534→        messages.push(json!({ "role": "user", "content": query }));
   535→    }
   536→    out
   537→}

   553→fn extract_user_query(body: &serde_json::Value) -> String {
   554→    let messages = match body.get("messages").and_then(|m| m.as_array()) {
   555→        Some(m) => m,
   556→        None => return String::new(),
   557→    };
   558→    for m in messages.iter().rev() {
   559→        if m.get("role").and_then(|r| r.as_str()) == Some("user") {
   560→            if let Some(c) = m.get("content") {
   561→                return content_to_string(c);
   562→            }
   563→        }
   564→    }
   565→    String::new()
   566→}
   567→
   568→fn content_to_string(c: &serde_json::Value) -> String {
   569→    match c {
   570→        serde_json::Value::String(s) => s.clone(),
   571→        serde_json::Value::Array(arr) => arr
   572→            .iter()
   573→            .filter_map(|p| {
   574→                if p.get("type").and_then(|t| t.as_str()) == Some("text") {
   575→                    p.get("text").and_then(|t| t.as_str()).map(|s| s.to_string())
   576→                } else {
   577→                    None
   578→                }
   579→            })
   580→            .collect::<Vec<_>>()
   581→            .join(" "),
   582→        _ => String::new(),
   583→    }
   584→}

   621→fn randomish_id() -> String {
   622→    use std::time::{SystemTime, UNIX_EPOCH};
   623→    let nanos = SystemTime::now()
   624→        .duration_since(UNIX_EPOCH)

   627→    format!("{:x}", nanos)

   638→/// Speculative prefetch: warm the retrieval cache for graph-adjacent candidate
   639→/// queries so the next, related query is instant.
   640→async fn prefetch(state: AppState, retrieved: &[ScoredNode]) {
   641→    if retrieved.is_empty() {
   642→        return;
   643→    }
   644→    let graph = state.graph.clone();
   645→    let cache = state.cache.clone();
   646→    let candidates: Vec<String> = retrieved.iter().take(3).map(|n| n.text.clone()).collect();
   647→    tokio::spawn(async move {
   648→        for q in candidates {
   649→            let (qvec, ctx) = {
   650→                let g = graph.lock().await;
   651→                let qv = g.vectorizer.vectorize(&q);
   652→                let r = g.retrieve(&q, 8);
   653→                let c = compress::compress(&r, &q, 6000);
   654→                (qv, c)
   655→            };
   656→            cache.lock().await.put_retrieval(&q, qvec, ctx);
   657→        }
   658→    });
   659→}

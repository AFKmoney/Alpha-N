     1→/**
     2→ * generated-app-renderer.tsx — runtime execution surface for AI-generated apps.
     3→ *
     4→ * When the user installs an app from the AI App Store, the LLM-produced
     5→ * React component source code is persisted to the `GeneratedApp` table.
     6→ * Opening the app spawns a "custom" window whose `data.generatedAppId`
     7→ * points at that row. This component:
     8→ *
     9→ *   1. Reads the generated-app id from the window's data.
    10→ *   2. Fetches the source code from /api/alpha/generate-app?id=<id>.
    11→ *   3. Displays the description + a syntax-highlighted source viewer.
    12→ *   4. On "Run", compiles the code via `new Function(...)` (sandboxed — only
    13→ *      the modules listed in the API's RUNTIME_CONTRACT are injected) and
    14→ *      renders it inside an error boundary.
    15→ *
    16→ * This is a sandboxed OS environment, so `new Function` execution is
    17→ * acceptable. The boundary catches render-time errors and shows a clear
    18→ * fallback so the rest of the OS keeps running.
    19→ */

   123→// ----------------------------------------------------------------------------
   124→// compileGeneratedCode — turn the LLM's source string into a runnable
   125→// React component. Uses Babel to strip TypeScript types and transpile JSX
   126→// to React.createElement calls, then wraps the result in `new Function`
   127→// with a curated module registry injected as the `require` parameter.
   128→// ----------------------------------------------------------------------------
   129→function compileGeneratedCode(code: string): CompiledApp {
   130→  // 1. Strip the "use client" directive (meaningful to bundlers, not us).
   131→  const stripped = code.replace(/^["']use client["'];?\s*\n?/m, "");
   132→
   133→  // 2. Transpile with Babel: TypeScript types stripped, JSX → createElement.
   134→  //    Babel 8 infers JSX parsing from the .tsx filename automatically (the
   135→  //    old `allExtensions`/`isTSX` options were removed).
   136→  let transformed: string;
   137→  try {
   138→    const out = Babel.transform(stripped, {
   139→      presets: ["typescript", ["react", { runtime: "classic" }]],
   140→      plugins: ["transform-modules-commonjs"],
   141→      filename: "generated.tsx",
   142→      sourceType: "module",
   143→    });
   144→    if (!out || typeof out.code !== "string") {
   145→      return { Component: null, error: "Babel transform returned no output." };
   146→    }
   147→    transformed = out.code;
   148→  } catch (err) {
   149→    const message = err instanceof Error ? err.message : String(err);
   150→    return {
   151→      Component: null,
   152→      error: `Babel transform failed: ${message.split("\n")[0]}`,
   153→    };
   154→  }
   155→
   156→  // 3. The transform-modules-commonjs plugin converts ESM imports into
   157→  //    `require("<source>")` calls and `export function X` into
   158→  //    `exports.X = function X`. We provide a `require` shim that resolves
   159→  //    against MODULE_REGISTRY, and a `module`/`exports` pair to capture
   160→  //    the exports.
   161→  const moduleObj: { exports: Record<string, unknown> } = { exports: {} };
   162→  const requireFn = (name: string): Record<string, unknown> => {
   163→    const mod = MODULE_REGISTRY[name];
   164→    if (!mod) {
   165→      throw new Error(`Unknown module requested by generated app: ${name}`);
   166→    }
   167→    return mod;
   168→  };
   169→
   170→  // 4. Compile via `new Function`. The transformed code is plain JS that
   171→  //    uses `require`, `exports`, `module`, and `React` (for createElement).
   172→  type RequireFn = (name: string) => unknown;
   173→  type ModuleObj = { exports: Record<string, unknown> };
   174→  type ReactNamespace = typeof import("react");
   175→  let factory: (module: ModuleObj, exports: Record<string, unknown>, require: RequireFn, React: ReactNamespace) => void;
   176→  try {
   177→    factory = new Function(
   178→      "module",
   179→      "exports",
   180→      "require",
   181→      "React",
   182→      `"use strict";\n${transformed}`
   183→    ) as typeof factory;
   184→  } catch (err) {
   185→    const message = err instanceof Error ? err.message : String(err);
   186→    return { Component: null, error: `Compile error: ${message}` };
   187→  }
   188→
   189→  // 5. Execute the factory to populate module.exports.
   190→  try {
   191→    factory(moduleObj, moduleObj.exports, requireFn, React);
   192→  } catch (err) {
   193→    const message = err instanceof Error ? err.message : String(err);
   194→    return { Component: null, error: `Module init error: ${message}` };
   195→  }
   196→
   197→  // 6. Find the exported component. Try common names, then fall back to the
   198→  //    first function-valued export.
   199→  const exports = moduleObj.exports;
   200→  const candidates = Object.values(exports).filter(
   201→    (v): v is React.ComponentType<{ windowId?: string }> => typeof v === "function"
   202→  );
   203→  if (candidates.length === 0) {
   204→    return {
   205→      Component: null,
   206→      error: "No React component was exported by the generated code.",
   207→    };
   208→  }
   209→
   210→  // Prefer the export whose name starts with an uppercase letter (React
   211→  // convention for components). Fall back to the first function otherwise.
   212→  const namedExport = Object.entries(exports).find(
   213→    ([name, v]) => typeof v === "function" && /^[A-Z]/.test(name)
   214→  );
   215→  const Component = (namedExport?.[1] ?? candidates[0]) as React.ComponentType<{
   216→    windowId?: string;
   217→  }>;
   218→  return { Component, error: null };
   219→}

   221→// ----------------------------------------------------------------------------
   222→// RuntimeErrorBoundary — catches render-time errors thrown by the generated
   223→// component so a buggy AI app can't take down the whole OS.
   224→// ----------------------------------------------------------------------------
   225→class RuntimeErrorBoundary extends React.Component<
   226→  RuntimeErrorBoundaryProps,
   227→  RuntimeErrorBoundaryState
   228→> {
   229→  constructor(props: RuntimeErrorBoundaryProps) {
   230→    super(props);
   231→    this.state = { error: null, retryNonce: 0 };
   232→  }
   233→
   234→  static getDerivedStateFromError(error: Error): RuntimeErrorBoundaryState {
   235→    return { error, retryNonce: 0 };
   236→  }
   237→
   238→  handleRetry = (): void => {
   239→    this.setState((s) => ({ error: null, retryNonce: s.retryNonce + 1 }));
   240→  };
   241→
   242→  render(): React.ReactNode {
   243→    if (this.state.error) {
   244→      return this.props.fallback(this.state.error, this.handleRetry);
   245→    }
   246→    // key by retryNonce so children remount when retrying
   247→    return (
   248→      <div key={this.state.retryNonce} className="h-full w-full">
   249→        {this.props.children}
   250→      </div>
   251→    );
   252→  }
   253→}

   255→// ----------------------------------------------------------------------------
   256→// Simple syntax highlighter — adapted from real-code-editor.tsx so the
   257→// generated source is readable in the Alpha-N aesthetic.
   258→// ----------------------------------------------------------------------------
   259→interface Token {
   260→  text: string;
   261→  cls: string;
   262→}
   263→
   264→const KEYWORDS = new Set([
   265→  "import", "export", "from", "class", "const", "let", "var", "function",
   266→  "async", "await", "if", "else", "return", "for", "while", "new", "this",
   267→  "type", "interface", "extends", "implements", "public", "private",
   268→  "protected", "readonly", "static", "void", "null", "undefined", "true",
   269→  "false", "as", "in", "of", "try", "catch", "finally", "throw", "switch",
   270→  "case", "default", "break", "continue", "enum", "typeof", "instanceof",
   271→]);
   272→
   273→function highlightLine(line: string): Token[] {
   274→  const tokens: Token[] = [];
   275→  const re =
   276→    /(\/\/[^\n]*)|('(?:[^'\\]|\\.)*'|"(?:[^"\\]|\\.)*"|`(?:[^`\\]|\\.)*`)|(\d+\.?\d*)|([A-Za-z_$][A-Za-z0-9_$]*)|(\s+)|([{}()\[\].,;:<>+\-*/%=&|!?]+)/g;
   277→  let m: RegExpExecArray | null;
   278→  while ((m = re.exec(line)) !== null) {
   279→    const [, comment, str, num, ident, ws, punct] = m;
   280→    if (comment !== undefined) tokens.push({ text: comment, cls: "text-muted-foreground/50 italic" });
   281→    else if (str !== undefined) tokens.push({ text: str, cls: "text-[oklch(0.7_0.18_145)]" });
   282→    else if (num !== undefined) tokens.push({ text: num, cls: "text-[oklch(0.85_0.14_55)]" });
   283→    else if (ident !== undefined) {
   284→      if (KEYWORDS.has(ident)) tokens.push({ text: ident, cls: "text-[oklch(0.74_0.22_300)]" });
   285→      else if (/^[A-Z]/.test(ident)) tokens.push({ text: ident, cls: "text-[oklch(0.82_0.16_85)]" });
   286→      else {
   287→        const rest = line.slice(re.lastIndex);
   288→        const isFn = /^\s*\(/.test(rest);
   289→        tokens.push({
   290→          text: ident,
   291→          cls: isFn ? "text-[oklch(0.82_0.17_195)]" : "text-foreground",
   292→        });
   293→      }
   294→    } else if (ws !== undefined) tokens.push({ text: ws, cls: "" });
   295→    else if (punct !== undefined) tokens.push({ text: punct, cls: "text-muted-foreground/70" });
   296→  }
   297→  return tokens.length ? tokens : [{ text: " ", cls: "" }];
   298→}

   329→function CodeViewer({ code }: { code: string }): React.ReactElement {
   330→  const [copied, setCopied] = useState(false);
   331→  const lines = code.split("\n");
   332→
   333→  const handleCopy = useCallback(() => {
   334→    try {
   335→      void navigator.clipboard.writeText(code);
   336→      setCopied(true);
   337→      window.setTimeout(() => setCopied(false), 1400);
   338→    } catch {
   339→      /* clipboard may be unavailable in some browsers */
   340→    }
   341→  }, [code]);
   342→
   343→  return (
   344→    <div className="relative flex min-h-0 flex-1 flex-col overflow-hidden rounded-lg border border-border/40 bg-[oklch(0.09_0.012_265)]">
   345→      <div className="flex items-center justify-between border-b border-border/40 px-3 py-1.5">
   346→        <span className="font-mono-ae text-[0.65rem] text-muted-foreground">
   347→          source · {lines.length} lines
   348→        </span>
   349→        <button
   350→          onClick={handleCopy}
   351→          className="flex items-center gap-1 rounded px-2 py-1 font-mono-ae text-[0.65rem] text-muted-foreground transition-colors hover:bg-foreground/[0.06] hover:text-foreground"
   352→        >
   353→          {copied ? (
   354→            <>
   355→              <Check className="h-3 w-3" /> copied
   356→            </>
   357→          ) : (
   358→            <>
   359→              <Copy className="h-3 w-3" /> copy
   360→            </>
   361→          )}
   362→        </button>
   363→      </div>
   364→      <div className="scroll-ae min-h-0 flex-1 overflow-auto">
   365→        <pre className="font-mono-ae text-[0.7rem] leading-relaxed">
   366→          <code className="block">
   367→            {lines.map((line, i) => (
   368→              <div
   369→                key={i}
   370→                className="flex hover:bg-foreground/[0.03]"
   371→              >
   372→                <span className="select-none border-r border-border/30 px-2 text-right text-muted-foreground/40 w-10 shrink-0">
   373→                  {i + 1}
   374→                </span>
   375→                <span className="px-3 whitespace-pre">
   376→                  {highlightLine(line).map((tok, j) => (
   377→                    <span key={j} className={tok.cls}>
   378→                      {tok.text}
   379→                    </span>
   380→                  ))}
   381→                </span>
   382→              </div>
   383→            ))}
   384→          </code>
   385→        </pre>
   386→      </div>
   387→    </div>
   388→  );
   389→}

   391→// ----------------------------------------------------------------------------
   392→// GeneratingState — shows a "AI is coding..." animation with fake streaming
   393→// code lines while the LLM generates the app. Listens for the
   394→// alpha-generated-app-ready / alpha-generated-app-error events.
   395→// ----------------------------------------------------------------------------
   396→function GeneratingState({
   397→  windowId,
   398→  tempId,
   399→  appName,
   400→  description,
   401→}: {
   402→  windowId: string;
   403→  tempId: string;
   404→  appName: string;
   405→  description: string;
   406→}): React.ReactElement {
   407→  const { setWindowData } = useOS();
   408→  const [streamedLines, setStreamedLines] = useState<string[]>([]);
   409→  const [done, setDone] = useState(false);
   410→
   411→  // Fake streaming code effect — shows lines appearing one by one to give
   412→  // the user the feeling that the AI is writing code in real-time.
   413→  const fakeCode = useMemo(
   414→    () => [
   415→      `"use client";`,
   416→      ``,
   417→      `import { useState, useEffect, useRef } from "react";`,
   418→      `import { Button } from "@/components/ui/button";`,
   419→      `import { Card, CardContent } from "@/components/ui/card";`,
   420→      ``,
   421→      `export function ${appName.replace(/[^A-Za-z0-9]/g, "") || "GeneratedApp"}({ windowId }: { windowId?: string }) {`,
   422→      `  const [state, setState] = useState(0);`,
   423→      `  // ${description.slice(0, 60)}`,
   424→      `  // ... AI is writing ...`,
   425→    ],
   426→    [appName, description]
   427→  );
   428→
   429→  useEffect(() => {
   430→    let i = 0;
   431→    const interval = setInterval(() => {
   432→      if (i < fakeCode.length) {
   433→        setStreamedLines((prev) => [...prev, fakeCode[i]]);
   434→        i++;
   435→      } else {
   436→        clearInterval(interval);
   437→      }
   438→    }, 200);
   439→    return () => clearInterval(interval);
   440→  }, [fakeCode]);
   441→
   442→  // Listen for the "ready" event — swap to the real app.
   443→  useEffect(() => {
   444→    const onReady = (e: Event) => {
   445→      const detail = (e as CustomEvent).detail as {
   446→        tempId: string;
   447→        realId: string;
   448→        name: string;
   449→        code: string;
   450→      };
   451→      if (detail.tempId !== tempId) return;
   452→      setDone(true);
   453→      // Update the window data so GeneratedAppRenderer re-fetches the real app
   454→      setWindowData(windowId, {
   455→        generatedAppId: detail.realId,
   456→        spec: description,
   457→        isGenerating: false,
   458→      });
   459→    };
   460→    const onError = (e: Event) => {
   461→      const detail = (e as CustomEvent).detail as { tempId: string; error: string };
   462→      if (detail.tempId !== tempId) return;
   463→      setWindowData(windowId, {
   464→        generatedAppId: null,
   465→        spec: description,
   466→        isGenerating: false,
   467→        generateError: detail.error,
   468→      });
   469→    };
   470→    window.addEventListener("alpha-generated-app-ready", onReady);
   471→    window.addEventListener("alpha-generated-app-error", onError);
   472→    return () => {
   473→      window.removeEventListener("alpha-generated-app-ready", onReady);
   474→      window.removeEventListener("alpha-generated-app-error", onError);
   475→    };
   476→  }, [tempId, windowId, setWindowData, description]);
   477→
   478→  return (
   479→    <div className="flex h-full w-full flex-col bg-background">
   480→      {/* Header */}
   481→      <div className="flex items-center gap-2 border-b border-border/50 px-4 py-2.5">
   482→        <Loader2 className="h-4 w-4 animate-spin text-[oklch(0.82_0.17_195)]" />
   483→        <div className="min-w-0 flex-1">
   484→          <h3 className="truncate font-mono-ae text-sm font-semibold text-foreground">
   485→            {done ? "Installing..." : `Generating: ${appName}`}
   486→          </h3>
   487→          <p className="truncate text-[0.65rem] text-muted-foreground">
   488→            {done ? "App compiled — loading..." : "AI is writing code..."}
   489→          </p>
   490→        </div>
   491→      </div>
   492→
   493→      {/* Streaming code viewer */}
   494→      <div className="flex min-h-0 flex-1 flex-col p-4">
   495→        <div className="rounded-lg border border-[oklch(0.82_0.17_195)]/20 bg-[oklch(0.09_0.012_265)] p-3">
   496→          <div className="mb-2 flex items-center gap-2">
   497→            <span className="h-2 w-2 animate-pulse rounded-full bg-[oklch(0.82_0.17_195)]" />
   498→            <span className="font-mono-ae text-[0.65rem] text-[oklch(0.82_0.17_195)]">
   499→              {done ? "compiling..." : "streaming source..."}
   500→            </span>
   501→          </div>
   502→          <pre className="font-mono-ae text-[0.7rem] leading-relaxed">
   503→            {streamedLines.map((line, i) => (
   504→              <div key={i} className="flex hover:bg-foreground/[0.03]">
   505→                <span className="select-none w-8 shrink-0 pr-2 text-right text-muted-foreground/40">
   506→                  {i + 1}
   507→                </span>
   508→                <span className="whitespace-pre text-foreground/70">{line}</span>
   509→              </div>
   510→            ))}
   511→            {!done && (
   512→              <span className="inline-block h-3.5 w-1.5 animate-pulse bg-[oklch(0.82_0.17_195)]" />
   513→            )}
   514→          </pre>
   515→        </div>
   516→
   517→        {/* Description */}
   518→        <div className="mt-3 rounded-lg border border-border/40 bg-card/30 p-3">
   519→          <div className="eyebrow mb-1">app description</div>
   520→          <p className="text-[0.75rem] leading-snug text-foreground">{description}</p>
   521→        </div>
   522→
   523→        {done && (
   524→          <motion.div
   525→            initial={{ opacity: 0 }}
   526→            animate={{ opacity: 1 }}
   527→            className="mt-3 flex items-center gap-2 rounded-lg border border-[oklch(0.7_0.18_145)]/30 bg-[oklch(0.7_0.18_145)]/[0.06] p-3"
   528→          >
   529→            <Check className="h-4 w-4 text-[oklch(0.7_0.18_145)]" />
   530→            <span className="font-mono-ae text-[0.7rem] text-[oklch(0.7_0.18_145)]">
   531→              Code generated — installing on desktop...
   532→            </span>
   533→          </motion.div>
   534→        )}
   535→      </div>
   536→    </div>
   537→  );

   540→// ----------------------------------------------------------------------------
   541→// Main component
   542→// ----------------------------------------------------------------------------
   543→export function GeneratedAppRenderer({
   544→  windowId,
   545→}: {
   546→  windowId: string;
   547→}): React.ReactElement {
   548→  const { windows, setWindowData } = useOS();
   549→  const win = windows.find((w) => w.id === windowId);
   550→  const generatedAppId = (win?.data?.generatedAppId as string) ?? null;
   551→  const spec = (win?.data?.spec as string) ?? "";
   552→  const isGenerating = (win?.data?.isGenerating as boolean) ?? false;
   553→  const generateError = (win?.data?.generateError as string) ?? null;
   554→  const suggestionName = (win?.data?.suggestionName as string) ?? "Generated App";
   555→  const suggestionDesc = (win?.data?.suggestionDesc as string) ?? spec;
   556→
   557→  const [loading, setLoading] = useState(true);
   558→  const [error, setError] = useState<string | null>(null);
   559→  const [app, setApp] = useState<GeneratedAppData | null>(null);
   560→  const [liveMode, setLiveMode] = useState(false);
   561→
   562→  // Fetch the generated app's source code.
   563→  const loadApp = useCallback(async (id: string): Promise<void> => {
   564→    setLoading(true);
   565→    setError(null);
   566→    try {
   567→      const r = await fetch(`/api/alpha/generate-app?id=${encodeURIComponent(id)}`);
   568→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   569→      const data = (await r.json()) as { ok: boolean; app?: GeneratedAppData; error?: string };
   570→      if (data.ok && data.app) {
   571→        setApp(data.app);
   572→        // Auto-run the app once loaded — the user wants to see it immediately.
   573→        setLiveMode(true);
   574→      } else {
   575→        setError(data.error || "App not found.");
   576→      }
   577→    } catch (err: unknown) {
   578→      setError(err instanceof Error ? err.message : "Network error.");
   579→    } finally {
   580→      setLoading(false);
   581→    }
   582→  }, []);
   583→
   584→  // Kick off the fetch whenever the id changes (and it's not a temp/generating id).
   585→  useEffect(() => {
   586→    if (!generatedAppId || isGenerating) return;
   587→    if (generatedAppId.startsWith("generating-")) return;
   588→    void loadApp(generatedAppId);
   589→  }, [generatedAppId, isGenerating, loadApp]);
   590→
   591→  // Pull `code` out so useMemo's dependency matches the inferred one.
   592→  const code = app?.code ?? "";
   593→
   594→  // Compile the source once per code change.
   595→  const compiled = useMemo<CompiledApp>(() => {
   596→    if (!code) return { Component: null, error: null };
   597→    return compileGeneratedCode(code);
   598→  }, [code]);
   599→
   600→  // -------- Render --------
   601→
   602→  // If the AI is still generating, show the streaming-code generating state.
   603→  if (isGenerating && generatedAppId && generatedAppId.startsWith("generating-")) {
   604→    return (
   605→      <GeneratingState
   606→        windowId={windowId}
   607→        tempId={generatedAppId}
   608→        appName={suggestionName}
   609→        description={suggestionDesc}
   610→      />
   611→    );
   612→  }
   613→
   614→  // If generation failed, show the error.
   615→  if (generateError) {
   616→    return <ErrorState message={generateError} />;
   617→  }
   618→
   619→  if (!generatedAppId) {
   620→    return <ErrorState message="No app id was attached to this window." />;
   621→  }
   622→  if (loading) return <LoadingState />;
   623→  if (error) return <ErrorState message={error} />;
   624→  if (!app) return <ErrorState message="App data is missing." />;
   625→
   626→  return (
   627→    <div className="flex h-full w-full flex-col bg-background">
   628→      {/* Header */}
   629→      <div className="flex items-center justify-between gap-2 border-b border-border/50 px-4 py-2.5">
   630→        <div className="min-w-0 flex-1">
   631→          <h3 className="truncate font-mono-ae text-sm font-semibold text-foreground">
   632→            {app.name}
   633→          </h3>
   634→          <p className="truncate text-[0.65rem] text-muted-foreground">
   635→            {app.category} · generated app
   636→          </p>
   637→        </div>
   638→        <div className="flex shrink-0 items-center gap-1.5">
   639→          {liveMode ? (
   640→            <button
   641→              onClick={() => setLiveMode(false)}
   642→              className="flex items-center gap-1.5 rounded-md border border-border/50 bg-card/50 px-2.5 py-1.5 font-mono-ae text-[0.7rem] text-foreground transition-colors hover:bg-card"
   643→            >
   644→              <Code2 className="h-3.5 w-3.5" />
   645→              View Source
   646→            </button>
   647→          ) : (
   648→            <button
   649→              onClick={() => setLiveMode(true)}
   650→              disabled={!compiled.Component}
   651→              className={cn(
   652→                "flex items-center gap-1.5 rounded-md border px-2.5 py-1.5 font-mono-ae text-[0.7rem] transition-colors disabled:opacity-40",
   653→                compiled.Component
   654→                  ? "border-[oklch(0.82_0.17_195)]/40 bg-[oklch(0.82_0.17_195)]/10 text-[oklch(0.82_0.17_195)] hover:bg-[oklch(0.82_0.17_195)]/20"
   655→                  : "border-border/50 bg-card/30 text-muted-foreground"
   656→              )}
   657→            >
   658→              <Play className="h-3.5 w-3.5" />
   659→              Run
   660→            </button>
   661→          )}
   662→        </div>
   663→      </div>
   664→
   665→      {/* Compile error banner — shown above either view */}
   666→      {compiled.error && (
   667→        <div className="flex items-start gap-2 border-b border-[oklch(0.78_0.2_20)]/30 bg-[oklch(0.78_0.2_20)]/[0.06] px-4 py-2">
   668→          <AlertCircle className="mt-0.5 h-3.5 w-3.5 shrink-0 text-[oklch(0.78_0.2_20)]" />
   669→          <div className="min-w-0 flex-1">
   670→            <p className="font-mono-ae text-[0.65rem] font-semibold text-[oklch(0.78_0.2_20)]">
   671→              compile error
   672→            </p>
   673→            <p className="mt-0.5 break-words text-[0.65rem] leading-snug text-muted-foreground">
   674→              {compiled.error}
   675→            </p>
   676→          </div>
   677→        </div>
   678→      )}
   679→
   680→      {/* Body */}
   681→      <div className="flex min-h-0 flex-1 flex-col gap-3 p-4">
   682→        {liveMode && compiled.Component ? (
   683→          <RuntimeErrorBoundary
   684→            fallback={(err: Error, retry: () => void) => (
   685→              <div className="flex h-full w-full flex-col items-center justify-center gap-3 p-6 text-center">
   686→                <motion.div
   687→                  animate={{ rotate: [0, -8, 8, 0] }}
   688→                  transition={{ duration: 0.6, repeat: Infinity, repeatDelay: 1.5 }}
   689→                >
   690→                  <AlertCircle className="h-8 w-8 text-[oklch(0.78_0.2_20)]" />
   691→                </motion.div>
   692→                <p className="font-mono-ae text-xs text-[oklch(0.78_0.2_20)]">
   693→                  runtime error
   694→                </p>
   695→                <p className="max-w-md break-words text-[0.7rem] leading-snug text-muted-foreground">
   696→                  {err.message}
   697→                </p>
   698→                <button
   699→                  onClick={retry}
   700→                  className="mt-1 flex items-center gap-1.5 rounded-md border border-border/50 bg-card/50 px-2.5 py-1.5 font-mono-ae text-[0.7rem] text-foreground transition-colors hover:bg-card"
   701→                >
   702→                  <RefreshCw className="h-3.5 w-3.5" />
   703→                  Retry
   704→                </button>
   705→              </div>
   706→            )}
   707→          >
   708→            <div className="h-full w-full overflow-hidden">
   709→              <compiled.Component windowId={windowId} />
   710→            </div>
   711→          </RuntimeErrorBoundary>
   712→        ) : (
   713→          <>
   714→            {/* Description */}
   715→            <div className="rounded-lg border border-[oklch(0.82_0.17_195)]/20 bg-[oklch(0.82_0.17_195)]/[0.04] p-3">
   716→              <div className="eyebrow mb-1">description</div>
   717→              <p className="text-[0.75rem] leading-snug text-foreground">
   718→                {app.description || spec || "No description provided."}
   719→              </p>
   720→            </div>
   721→
   722→            {/* Source viewer */}
   723→            <CodeViewer code={app.code} />
   724→
   725→            {/* Footer hint */}
   726→            <div className="flex items-center justify-between gap-2 px-1">
   727→              <p className="font-mono-ae text-[0.6rem] text-muted-foreground/60">
   728→                sandboxed via <span className="text-[oklch(0.82_0.17_195)]">new Function()</span>

   732→                  ✓ compiles cleanly

   735→                <p className="font-mono-ae text-[0.6rem] text-[oklch(0.78_0.2_20)]">
   736→                  ✗ cannot compile
   737→                </p>
   738→              )}
   739→            </div>
   740→          </>
   741→        )}
   742→      </div>
   743→    </div>
   744→  );

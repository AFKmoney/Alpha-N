   265→function extractJson(text: string): unknown {
   266→  let t = text.trim();
   267→  const fence = String.fromCharCode(96, 96, 96); // "```"
   268→  if (t.startsWith(fence)) {
   269→    t = t.replace(new RegExp("^" + fence + "(?:json)?\\s*", "i"), "").replace(new RegExp(fence + "\\s*$", "i"), "").trim();
   270→  }
   271→  const start = t.indexOf("{");
   272→  const end = t.lastIndexOf("}");
   273→  if (start === -1 || end === -1) throw new Error("no JSON object found");
   274→  return JSON.parse(t.slice(start, end + 1));
   275→}

   277→/**
   278→ * When the model fails to emit valid JSON (common with small local models on
   279→ * a strict schema), recover a usable free-text reply WITHOUT leaking the
   280→ * echoed system prompt into the chat.
   281→ *
   282→ * Strategy:
   283→ *   1. Drop everything before the last recognisable reply marker
   284→ *      ("message":, assistant:, N-CORE:) — the model's actual answer tends
   285→ *      to follow these.
   286→ *   2. Strip JSON fragments, code fences, and section headers copied from
   287→ *      the prompt ("# AKASHA", "### ", etc.).
   288→ *   3. Collapse repeated prompt echoes (lines that are obviously the system
   289→ *      prompt verbatim).
   290→ *   4. Return the cleaned text, capped. If nothing usable remains, return
   291→ *      an empty string so the chat shows nothing rather than garbage — the
   292→ *      cycle still logs the parse failure.
   293→ */
   294→function extractReplyFromRaw(raw: string): string {
   295→  let t = raw.trim();
   296→  if (!t) return "";
   297→
   298→  // 1. Try to anchor on the most recent reply marker the model tends to emit.
   299→  const markers = [
   300→    /"message"\s*:\s*"([^"]*(?:\\.[^"]*)*)"/i, // "message": "..." (single-line ok)
   301→    /\bassistant\s*:\s*([\s\S]+)/i,
   302→    /\bN-CORE\s*:\s*([\s\S]+)/i,
   303→    /\bREPLY\s*:\s*([\s\S]+)/i,
   304→  ];
   305→  for (const re of markers) {
   306→    const m = t.match(re);
   307→    if (m && m[1] && m[1].trim().length > 2) {
   308→      t = m[1].trim();
   309→      break;
   310→    }
   311→  }
   312→
   313→  // 2. Strip code fences and obvious JSON-object fragments.
   314→  t = t.replace(/```[\s\S]*?```/g, " "); // remove fenced blocks
   315→  t = t.replace(/^\s*[\[{][\s\S]*[\]}]\s*$/, " "); // whole-string JSON blob
   316→  t = t.replace(/\\n/g, " ").replace(/\\"/g, '"');
   317→
   318→  // 3. Remove prompt-section headers and prompt echoes the model copied.
   319→  const echoPatterns = [
   320→    /^\s*#+\s.*$/gm, // markdown headers ("# AKASHA", "### FEEDBACK...")
   321→    /^\s*(USER|N-CORE|SYSTEM|ASSISTANT)\s*:/gim,
   322→    /^\s*You are Alpha-OS\b.*/gim,
   323→    /^\s*MUTATIONS AVAILABLE\b.*/gim,
   324→    /^\s*"type"\s*:\s*"[a-z_]+"[\s\S]*$/gm, // trailing mutation JSON lines
   325→    /^\s*\{[\s\S]*$/gm, // dangling object openings
   326→  ];
   327→  for (const re of echoPatterns) t = t.replace(re, " ");
   328→
   329→  // 4. Collapse whitespace, cap length.
   330→  t = t.replace(/\s+/g, " ").trim();
   331→  if (t.length > 400) t = t.slice(0, 400).trim() + "…";
   332→
   333→  // If what's left is too short or looks like pure prompt residue, show
   334→  // nothing rather than confuse the user.
   335→  if (t.length < 3) return "";
   336→  return t;
   337→}

   428→    // ---- Learning loop: what worked, what failed, what to do now ----

   479→

   535→    // Retry with exponential backoff for 429 (rate limit) / network errors.
   536→    // Uses the universal callLLM which routes to cloud (z-ai SDK) or local

   538→    let raw = "";
   539→    let lastError: Error | null = null;
   540→    for (let attempt = 0; attempt < 3; attempt++) {
   541→      try {
   542→        const response = await callLLM(fullSystemPrompt, stateText, body.screenshot);
   543→        raw = response.content;
   544→        lastError = null;
   545→        break;
   546→      } catch (err) {
   547→        lastError = err instanceof Error ? err : new Error(String(err));
   548→        const msg = lastError.message;
   549→        if (msg.includes("429") || msg.includes("Too many requests") || msg.includes("fetch failed") || msg.includes("ECONNRESET") || msg.includes("Local LLM error")) {
   550→          const delay = Math.pow(2, attempt + 1) * 1000;
   551→          await new Promise((r) => setTimeout(r, delay));
   552→          continue;
   553→        }
   554→        break;
   555→      }
   556→    }

   561→      return NextResponse.json({
   562→        error: message,
   563→        rateLimited: isRateLimit,
   564→        reasoning: isRateLimit
   565→          ? "The cognitive API is rate-limiting me. I will wait longer before the next cycle."
   566→          : "The cognitive layer threw an exception.",

   593→        ],
   594→      };
   595→    }

   602→    });
   603→  } catch (err) {
   604→    const message = err instanceof Error ? err.message : "unknown error";
   605→    return NextResponse.json(
   606→      {
   607→        error: message,
   608→        rateLimited: false,
   609→        reasoning: "The cognitive layer threw an exception.",
   610→        message: "My cognitive layer hit an error. I will retry on the next beat.",
   611→        mutations: [], // NO fallback mutations during errors
   612→      },
   613→      { status: 200 }
   614→    );
   615→  }

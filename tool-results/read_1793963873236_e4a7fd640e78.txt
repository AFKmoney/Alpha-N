    24→// ============ BROWSER APP (with proxy — works on ANY site including google.com) ============
    25→export function BrowserApp({ windowId }: { windowId: string }) {
    26→  const { windows, setWindowData } = useOS();
    27→  const win = windows.find((w) => w.id === windowId);
    28→  const [input, setInput] = useState((win?.data?.url as string) ?? "https://example.com");
    29→  const [loading, setLoading] = useState(true);
    30→
    31→  const currentUrl = (win?.data?.url as string) ?? "https://example.com";
    32→  // The proxy URL strips X-Frame-Options so any site loads in the iframe
    33→  const proxyUrl = `/api/alpha/proxy?url=${encodeURIComponent(currentUrl)}`;
    34→
    35→  const navigate = (url: string) => {
    36→    let u = url.trim();
    37→    if (!u) return;
    38→    // If it looks like a search query (no dots), use Google search
    39→    if (!/^https?:\/\//.test(u) && !u.includes(".")) {
    40→      u = "https://www.google.com/search?q=" + encodeURIComponent(u);
    41→    } else if (!/^https?:\/\//.test(u)) {
    42→      u = "https://" + u;
    43→    }
    44→    setWindowData(windowId, { url: u });
    45→    setInput(u);
    46→    setLoading(true);
    47→  };
    48→
    49→  return (
    50→    <div className="flex h-full flex-col bg-background">
    51→      <div className="flex items-center gap-2 border-b border-border/50 p-2">
    52→        <button
    53→          onClick={() => { setLoading(true); setWindowData(windowId, { url: currentUrl + (currentUrl.includes("?") ? "&" : "?") + "_r=" + Date.now() }); }}
    54→          className="rounded p-1 text-muted-foreground hover:bg-foreground/10"
    55→          aria-label="Reload"
    56→        >
    57→          <RotateCw className="h-3.5 w-3.5" />
    58→        </button>
    59→        <form
    60→          onSubmit={(e) => { e.preventDefault(); navigate(input); }}
    61→          className="flex flex-1 items-center gap-1.5 rounded-full border border-border/60 bg-card/40 px-3 py-1"
    62→        >
    63→          <Globe className="h-3 w-3 text-muted-foreground" />
    64→          <input
    65→            value={input}
    66→            onChange={(e) => setInput(e.target.value)}
    67→            className="min-w-0 flex-1 bg-transparent font-mono-ae text-xs text-foreground focus:outline-none"
    68→            placeholder="Enter URL or search…"
    69→          />
    70→        </form>
    71→      </div>
    72→      <div className="relative min-h-0 flex-1 bg-white">
    73→        {loading && (
    74→          <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur">
    75→            <div className="flex items-center gap-2">
    76→              <motion.div
    77→                animate={{ rotate: 360 }}
    78→                transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
    79→                className="h-4 w-4 border-2 border-[oklch(0.82_0.17_195)] border-t-transparent rounded-full"
    80→              />
    81→              <span className="font-mono-ae text-xs text-muted-foreground">loading {currentUrl}…</span>
    82→            </div>
    83→          </div>
    84→        )}
    85→        <iframe
    86→          key={proxyUrl}
    87→          src={proxyUrl}
    88→          className="h-full w-full border-0"
    89→          sandbox="allow-scripts allow-same-origin allow-forms allow-popups"
    90→          title="browser"
    91→          onLoad={() => setLoading(false)}
    92→        />
    93→      </div>
    94→    </div>
    95→  );
    96→}

    98→// ============ FILES APP (real — reads from the actual filesystem API) ============
    99→// Uses Alpha-OS terminology: folders = "sectors", files = "vectors".
   100→// Supports right-click on entries (rename, delete, move, copy) via the
   101→// context-menu system, plus a toolbar to create new sectors and vectors.
   102→export function FilesApp() {
   103→  const [currentPath, setCurrentPath] = useState("");
   104→  const [entries, setEntries] = useState<{ name: string; isDir: boolean }[]>([]);
   105→  const [loading, setLoading] = useState(true);
   106→  const [fileContent, setFileContent] = useState<string | null>(null);
   107→  const [viewingFile, setViewingFile] = useState<string | null>(null);
   108→
   109→  // Load directory listing for the current sector path.
   110→  const loadDir = useCallback(() => {
   111→    setLoading(true);
   112→    setFileContent(null);
   113→    setViewingFile(null);
   114→    void fetch(`/api/alpha/files?path=${encodeURIComponent(currentPath)}`)
   115→      .then((r) => r.json())
   116→      .then((data) => {
   117→        if (data.type === "dir") {
   118→          setEntries(data.entries || []);
   119→        } else if (data.type === "file") {
   120→          setFileContent(data.content || "");
   121→          setViewingFile(currentPath);
   122→        }
   123→        setLoading(false);
   124→      })
   125→      .catch(() => setLoading(false));
   126→  }, [currentPath]);
   127→
   128→  useEffect(() => {
   129→    // Defer to a microtask to avoid the react-hooks/set-state-in-effect rule.
   130→    Promise.resolve().then(() => loadDir());
   131→  }, [loadDir]);
   132→
   133→  // Listen for alpha-files-refresh (dispatched by the context menu after
   134→  // create/delete/rename/move/copy operations) and re-list the sector.
   135→  useEffect(() => {
   136→    const onRefresh = () => loadDir();
   137→    window.addEventListener("alpha-files-refresh", onRefresh);
   138→    return () => window.removeEventListener("alpha-files-refresh", onRefresh);
   139→  }, [loadDir]);
   140→
   141→  const protectedPaths = ["kernel/", "prisma/schema.prisma", ".env", "Caddyfile"];
   142→  const isProtected = (name: string) => protectedPaths.some((p) => name.startsWith(p) || name === p);
   143→
   144→  /** Build the full relative path for an entry in the current sector. */
   145→  const entryPath = (name: string) => (currentPath ? `${currentPath}/${name}` : name);
   146→
   147→  /** Create a new sector (directory) inside the current sector. */
   148→  const createSector = () => {
   149→    const name = window.prompt("New sector name:", "new-sector");
   150→    if (!name) return;
   151→    void fetch("/api/alpha/files", {
   152→      method: "POST",
   153→      headers: { "content-type": "application/json" },
   154→      body: JSON.stringify({ path: entryPath(name.replace(/\s+/g, "-")), action: "mkdir" }),
   155→    }).then(() => loadDir());
   156→  };
   157→
   158→  /** Create a new vector (empty file) inside the current sector. */
   159→  const createVector = () => {
   160→    const name = window.prompt("New vector name:", "new-vector.txt");
   161→    if (!name) return;
   162→    void fetch("/api/alpha/files", {
   163→      method: "POST",
   164→      headers: { "content-type": "application/json" },
   165→      body: JSON.stringify({ path: entryPath(name.replace(/\s+/g, "-")), action: "touch" }),
   166→    }).then(() => loadDir());
   167→  };
   168→
   169→  return (
   170→    <div className="flex h-full flex-col bg-background">
   171→      {/* Toolbar: back + breadcrumb + create buttons */}
   172→      <div className="flex items-center gap-2 border-b border-border/50 px-3 py-2">
   173→        <button
   174→          onClick={() => currentPath && setCurrentPath(currentPath.split("/").slice(0, -1).join("/"))}
   175→          className="rounded p-1 text-muted-foreground hover:bg-foreground/10 disabled:opacity-30"
   176→          disabled={!currentPath}
   177→          title="Up to parent sector"
   178→        >
   179→          <RotateCw className="h-3 w-3" style={{ transform: "rotate(-45deg)" }} />
   180→        </button>
   181→        <span className="flex-1 truncate font-mono-ae text-xs text-muted-foreground">
   182→          /alpha-n/{currentPath || ""}
   183→        </span>
   184→        <button
   185→          onClick={createSector}
   186→          className="flex items-center gap-1 rounded px-2 py-1 text-xs text-[oklch(0.82_0.17_195)] hover:bg-foreground/10"
   187→          title="Create new sector (folder)"
   188→        >
   189→          <FolderPlus className="h-3.5 w-3.5" />
   190→        </button>
   191→        <button
   192→          onClick={createVector}
   193→          className="flex items-center gap-1 rounded px-2 py-1 text-xs text-[oklch(0.85_0.16_85)] hover:bg-foreground/10"
   194→          title="Create new vector (file)"
   195→        >
   196→          <FilePlus className="h-3.5 w-3.5" />
   197→        </button>
   198→      </div>
   199→
   200→      {/* Vector content view (when a file is selected) */}
   201→      {viewingFile && fileContent !== null ? (
   202→        <div className="scroll-ae min-h-0 flex-1 overflow-auto p-3">
   203→          <div className="mb-2 flex items-center gap-2">
   204→            <span className="font-mono-ae text-xs text-[oklch(0.82_0.17_195)]">{viewingFile}</span>
   205→            <button
   206→              onClick={() => { setViewingFile(null); setFileContent(null); }}
   207→              className="rounded p-1 text-muted-foreground hover:bg-foreground/10"
   208→            >
   209→              <X className="h-3 w-3" />
   210→            </button>
   211→          </div>
   212→          <pre className="font-mono-ae text-[0.72rem] leading-relaxed text-foreground/80 whitespace-pre-wrap break-all">
   213→            {fileContent.slice(0, 10000)}
   214→          </pre>
   215→        </div>
   216→      ) : (
   217→        <div className="scroll-ae min-h-0 flex-1 overflow-y-auto p-2 font-mono-ae text-xs">
   218→          {loading ? (
   219→            <div className="flex h-full items-center justify-center">
   220→              <RotateCw className="h-4 w-4 animate-spin text-muted-foreground" />
   221→            </div>
   222→          ) : entries.length === 0 ? (
   223→            <p className="py-4 text-center text-muted-foreground/60">Empty sector — right-click the desktop to create vectors here</p>
   224→          ) : (
   225→            entries.map((entry) => (
   226→              <button
   227→                key={entry.name}
   228→                onClick={() => setCurrentPath(entryPath(entry.name))}
   229→                onContextMenu={(e) => {
   230→                  triggerContextMenu(e, buildFileActions({
   231→                    name: entry.name,
   232→                    path: entryPath(entry.name),
   233→                    isDir: entry.isDir,
   234→                    onOpen: () => setCurrentPath(entryPath(entry.name)),
   235→                    onRefresh: loadDir,
   236→                  }));
   237→                }}
   238→                className="flex w-full items-center gap-2 rounded px-2 py-1.5 text-left hover:bg-foreground/[0.06]"
   239→              >
   240→                <span className={entry.isDir ? "text-[oklch(0.82_0.17_195)]" : "text-muted-foreground"}>
   241→                  {entry.isDir ? "▸" : "·"}
   242→                </span>
   243→                <span className={entry.isDir ? "text-[oklch(0.82_0.17_195)]" : "text-foreground/80"}>
   244→                  {entry.name}
   245→                </span>
   246→                {isProtected(entry.name) && <Lock className="ml-auto h-3 w-3 text-[oklch(0.85_0.16_85)]" />}
   247→              </button>
   248→            ))
   249→          )}
   250→        </div>
   251→      )}
   252→    </div>
   253→  );
   254→}

   344→// ============ SECURITY FOUNDATION APP ============
   345→/**
   346→ * SecurityApp — displays REAL, LIVE security data from the OS.
   347→ *
   348→ * Pulls live data from multiple sources:
   349→ *   1. os-store:  protectedFiles (the kernel file list),
   350→ *                 violationAttempts (real blocked writes),
   351→ *                 snapshots (rollback points),
   352→ *                 rollbackEvents (actual rollbacks that happened).
   353→ *   2. evolution-store:  generation, version, metrics (cpu/ram/entropy/coherence),
   354→ *                        mutationStream (live mutation log — shows BLOCKED entries),
   355→ *                        compileResults (real tsc/eslint results),
   356→ *                        debateResults (real council verdicts),
   357→ *                        rewardModel (real helpful/harmful tracking).
   358→ *
   359→ * Nothing is hardcoded or simulated — every number and entry is live state.
   360→ */
   361→export function SecurityApp() {
   362→  // ---- Pull live data from both stores ----
   363→  const {
   364→    protectedFiles,
   365→    violationAttempts,
   366→    snapshots,
   367→    rollbackEvents,
   368→  } = useOS();
   369→
   370→  const {
   371→    generation,
   372→    version,
   373→    metrics,
   374→    mutationStream,
   375→    compileResults,
   376→    debateResults,
   377→    rewardModel,
   378→    aiState,
   379→    aiBusy,
   380→  } = useEvolution();
   381→
   382→  // ---- Derived live metrics ----
   383→  const blockedMutations = mutationStream.filter(
   384→    (m) => m.kind === "violation"
   385→  );
   386→  const compileErrors = compileResults.filter((c) => !c.ok);
   387→  const rejectedDebates = debateResults.filter(
   388→    (d) => d.consensus === "REJECT"
   389→  );
   390→  const harmfulMutations = rewardModel.filter((r) => !r.helpful);
   391→  const totalMutations = mutationStream.length;
   392→  const violationRate =
   393→    totalMutations > 0
   394→      ? ((blockedMutations.length / totalMutations) * 100).toFixed(1)
   395→      : "0.0";
   396→
   397→  // Overall security status
   398→  const securityScore = Math.max(
   399→    0,
   400→    100 -
   401→      violationAttempts.length * 5 -
   402→      blockedMutations.length * 3 -
   403→      compileErrors.length * 2 -
   404→      harmfulMutations.length
   405→  );
   406→
   407→  return (
   408→    <div className="scroll-ae flex h-full flex-col overflow-y-auto bg-background p-3">
   409→      {/* ---- Header ---- */}
   410→      <div className="mb-3 flex items-center gap-2">
   411→        <Shield className="h-4 w-4 text-[oklch(0.85_0.16_85)]" />
   412→        <h3 className="font-mono-ae text-sm font-semibold">Security Foundation</h3>
   413→        <span className="ml-auto font-mono-ae text-[0.6rem] text-muted-foreground">
   414→          gen {generation} · v{version}
   415→        </span>
   416→      </div>
   417→
   418→      {/* ---- Live Security Score ---- */}
   419→      <div className="mb-3 rounded-xl border border-border/50 bg-card/40 p-3">
   420→        <div className="flex items-center justify-between">
   421→          <span className="eyebrow">security score</span>
   422→          <span
   423→            className={cn(
   424→              "font-mono-ae text-lg font-bold",
   425→              securityScore > 80
   426→                ? "text-[oklch(0.7_0.18_145)]"
   427→                : securityScore > 50
   428→                  ? "text-[oklch(0.85_0.16_85)]"
   429→                  : "text-[oklch(0.78_0.2_20)]"
   430→            )}
   431→          >
   432→            {securityScore}
   433→          </span>
   434→        </div>
   435→        <div className="mt-1.5 h-2 overflow-hidden rounded-full bg-foreground/10">
   436→          <div
   437→            className={cn(
   438→              "h-full rounded-full transition-all",
   439→              securityScore > 80
   440→                ? "bg-[oklch(0.7_0.18_145)]"
   441→                : securityScore > 50
   442→                  ? "bg-[oklch(0.85_0.16_85)]"
   443→                  : "bg-[oklch(0.65_0.24_25)]"
   444→            )}
   445→            style={{ width: `${securityScore}%` }}
   446→          />
   447→        </div>
   448→      </div>
   449→
   450→      {/* ---- Live Metrics Grid ---- */}
   451→      <div className="mb-3 grid grid-cols-2 gap-2">
   452→        <MetricBox
   453→          label="violations"
   454→          value={violationAttempts.length}
   455→          color={violationAttempts.length > 0 ? "text-[oklch(0.78_0.2_20)]" : "text-[oklch(0.7_0.18_145)]"}
   456→        />
   457→        <MetricBox
   458→          label="blocked writes"
   459→          value={blockedMutations.length}
   460→          color={blockedMutations.length > 0 ? "text-[oklch(0.78_0.2_20)]" : "text-[oklch(0.7_0.18_145)]"}
   461→        />
   462→        <MetricBox
   463→          label="rollbacks"
   464→          value={rollbackEvents.length}

   828→        setError(data.error);

   880→          value={password}

   885→        />

     9→/**
    10→ * Resolve a Prisma client that has the `generatedApp` model.
    11→ *
    12→ * Why this exists: the shared `@/lib/db` singleton caches the client on
    13→ * `globalThis`, so if the dev server started before the GeneratedApp model
    14→ * was added to the schema, the cached client (and even the cached
    15→ * `PrismaClient` class itself) lacks that model. We detect the staleness
    16→ * and create a fresh client; if the fresh client still lacks the model
    17→ * (because the PrismaClient class itself is cached), we fall back to raw
    18→ * SQL helpers (`generatedAppRaw`) that work on any client.
    19→ */
    20→function getDb(): PrismaClient {
    21→  const g = globalThis as unknown as {
    22→    __alphaGeneratedAppDb?: PrismaClient;
    23→    prisma?: PrismaClient;
    24→  };
    25→  if (g.__alphaGeneratedAppDb) return g.__alphaGeneratedAppDb;
    26→  const shared = g.prisma;
    27→  if (
    28→    shared &&
    29→    typeof (shared as unknown as { generatedApp?: unknown }).generatedApp !==
    30→      "undefined"
    31→  ) {
    32→    g.__alphaGeneratedAppDb = shared;
    33→    return shared;
    34→  }
    35→  const fresh = new PrismaClient({ log: ["query"] });
    36→  g.__alphaGeneratedAppDb = fresh;
    37→  return fresh;
    38→}
    39→
    40→const db = getDb();

    43→ * Stale-client detector. True when the loaded PrismaClient class doesn't
    44→ * know about the GeneratedApp model (i.e. the dev server was started before

    46→ */
    47→const STALE_CLIENT =

   291→/**
   292→ * Extract a fenced code block from an LLM response. The LLM is told not to
   293→ * include fences, but it sometimes does anyway. Returns the raw code string.
   294→ */
   295→function extractCodeBlock(text: string): string {
   296→  // Match ```lang\n ... ``` (with optional language tag)
   297→  const fenced = text.match(/```(?:tsx?|jsx?|js|ts)?\s*\n([\s\S]*?)```/);
   298→  if (fenced && fenced[1]) {
   299→    return fenced[1].trim();
   300→  }
   301→  return text.trim();
   302→}
   303→
   304→/**
   305→ * Parse the JSON array of app suggestions from an LLM response. The LLM is
   306→ * told to return ONLY JSON, but it sometimes prepends commentary — we strip
   307→ * anything outside the first `[` and the last `]`.
   308→ */
   309→function parseSuggestions(text: string): AppSuggestion[] {
   310→  const start = text.indexOf("[");
   311→  const end = text.lastIndexOf("]");
   312→  if (start === -1 || end === -1 || end <= start) return [];
   313→  const slice = text.slice(start, end + 1);
   314→  try {
   315→    const parsed = JSON.parse(slice);
   316→    if (!Array.isArray(parsed)) return [];
   317→    return parsed
   318→      .filter(
   319→        (p): p is AppSuggestion =>
   320→          typeof p === "object" &&
   321→          p !== null &&
   322→          typeof p.name === "string" &&
   323→          typeof p.description === "string"
   324→      )
   325→      .map((p) => ({
   326→        name: p.name,
   327→        description: p.description,
   328→        icon: typeof p.icon === "string" && p.icon.length > 0 ? p.icon : "▢",
   329→        category:
   330→          typeof p.category === "string" && p.category.length > 0
   331→            ? p.category
   332→            : "Productivity",
   333→      }));
   334→  } catch {
   335→    return [];
   336→  }
   337→}

   339→/**
   340→ * Validate that generated code meets the minimum contract: has "use client"
   341→ * directive and exports a named function. Returns the error message or null
   342→ * if valid.
   343→ */
   344→function validateGeneratedCode(code: string): string | null {
   345→  if (code.length < 40) return "Generated code is suspiciously short.";
   346→  if (!/["']use client["']/.test(code)) {
   347→    return 'Generated code is missing the "use client" directive.';
   348→  }
   349→  // Accept either `export function Name` or `export const Name = ...`
   350→  if (!/export\s+(function|const)\s+[A-Z][A-Za-z0-9_]+/.test(code)) {
   351→    return "Generated code is missing a named export (export function/const).";
   352→  }
   353→  return null;
   354→}

   360→// GET — list installed apps, or fetch one by id
   361→export async function GET(req: NextRequest) {
   362→  const { searchParams } = new URL(req.url);
   363→  const id = searchParams.get("id");
   364→
   365→  try {
   366→    if (STALE_CLIENT) {
   367→      // Use raw SQL — the cached PrismaClient class doesn't know about
   368→      // GeneratedApp, but $queryRawUnsafe speaks to SQLite directly.
   369→      if (id) {
   370→        const app = await rawGetApp(id);
   371→        if (!app) {
   372→          return NextResponse.json({ error: "App not found" }, { status: 404 });
   373→        }
   374→        return NextResponse.json({ ok: true, app });
   375→      }
   376→      const apps = await rawListApps();
   377→      return NextResponse.json({ ok: true, apps });
   378→    }
   379→
   380→    if (id) {
   381→      const app = await db.generatedApp.findUnique({ where: { id } });
   382→      if (!app) {
   383→        return NextResponse.json({ error: "App not found" }, { status: 404 });
   384→      }
   385→      const row: GeneratedAppRow = {
   386→        id: app.id,
   387→        name: app.name,
   388→        description: app.description,
   389→        category: app.category,
   390→        code: app.code,
   391→        createdAt: app.createdAt.toISOString(),
   392→      };
   393→      return NextResponse.json({ ok: true, app: row });
   394→    }
   395→
   396→    const apps = await db.generatedApp.findMany({
   397→      orderBy: { createdAt: "desc" },
   398→    });
   399→    const rows: GeneratedAppRow[] = apps.map((a) => ({
   400→      id: a.id,
   401→      name: a.name,
   402→      description: a.description,
   403→      category: a.category,
   404→      code: a.code,
   405→      createdAt: a.createdAt.toISOString(),
   406→    }));
   407→    return NextResponse.json({ ok: true, apps: rows });
   408→  } catch (err) {
   409→    return NextResponse.json(
   410→      { error: err instanceof Error ? err.message : "unknown" },
   411→      { status: 500 }
   412→    );
   413→  }
   414→}

   416→// POST — suggest app ideas OR generate one
   417→export async function POST(req: NextRequest) {
   418→  try {
   419→    const body = await req.json();
   420→    const action = typeof body.action === "string" ? body.action : "generate";
   421→
   422→    // -------- action: suggest --------
   423→    if (action === "suggest") {
   424→      const category = typeof body.category === "string" ? body.category : "Productivity";
   425→      const safeCategory = isCategory(category) ? category : "Productivity";
   426→
   427→      const sys = buildSuggestSystemPrompt();
   428→      const user = buildSuggestUserPrompt(safeCategory);
   429→
   430→      let suggestions: AppSuggestion[] = [];
   431→      try {
   432→        const resp = await callLLM(sys, user);
   433→        suggestions = parseSuggestions(resp.content);
   434→      } catch (err) {
   435→        return NextResponse.json(
   436→          {
   437→            ok: false,
   438→            error: err instanceof Error ? err.message : "LLM call failed",
   439→            suggestions: [],
   440→          },
   441→          { status: 502 }
   442→        );
   443→      }
   444→
   445→      if (suggestions.length === 0) {
   446→        // Fall back to a curated set so the UI never shows an empty grid.
   447→        suggestions = fallbackSuggestions(safeCategory);
   448→      }
   449→
   450→      return NextResponse.json({ ok: true, suggestions });
   451→    }
   452→
   453→    // -------- action: install (AI provides code directly, no LLM call needed) --------
   454→    if (action === "install") {
   455→      const name = typeof body.name === "string" ? body.name.trim() : "";
   456→      const description = typeof body.description === "string" ? body.description.trim() : "";
   457→      const category = typeof body.category === "string" ? body.category : "AI Tools";
   458→      const code = typeof body.code === "string" ? body.code : "";
   459→
   460→      if (!name || !code) {
   461→        return NextResponse.json(
   462→          { ok: false, error: "name and code are required for install" },
   463→          { status: 400 }
   464→        );
   465→      }
   466→
   467→      const safeCategory = isCategory(category) ? category : "AI Tools";
   468→      const id = `ga-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;
   469→
   470→      try {
   471→        if (STALE_CLIENT) {
   472→          await rawCreateApp({ id, name, description, category: safeCategory, code });
   473→        } else {
   474→          await db.generatedApp.create({ data: { id, name, description, category: safeCategory, code } });
   475→        }
   476→        return NextResponse.json({ ok: true, app: { id, name, code } });
   477→      } catch (err) {
   478→        return NextResponse.json(
   479→          { ok: false, error: err instanceof Error ? err.message : "DB error" },
   480→          { status: 500 }
   481→        );
   482→      }
   483→    }
   484→
   485→    // -------- action: generate (default) --------
   486→    const name = typeof body.name === "string" ? body.name.trim() : "";
   487→    const description =
   488→      typeof body.description === "string" ? body.description.trim() : "";
   489→    const category = typeof body.category === "string" ? body.category : "Productivity";
   490→    const safeCategory = isCategory(category) ? category : "Productivity";
   491→
   492→    if (!name || !description) {
   493→      return NextResponse.json(
   494→        { ok: false, error: "name and description are required" },
   495→        { status: 400 }
   496→      );
   497→    }
   498→
   499→    const sys = buildGenerateSystemPrompt();
   500→    const user = buildGenerateUserPrompt(name, description, safeCategory);
   501→
   502→    let code = "";
   503→    try {
   504→      const resp = await callLLM(sys, user);
   505→      code = extractCodeBlock(resp.content);
   506→    } catch (err) {
   507→      return NextResponse.json(
   508→        { ok: false, error: err instanceof Error ? err.message : "LLM call failed" },
   509→        { status: 502 }
   510→      );
   511→    }
   512→
   513→    const validationError = validateGeneratedCode(code);
   514→    if (validationError) {
   515→      return NextResponse.json(
   516→        {
   517→          ok: false,
   518→          error: `Code validation failed: ${validationError}`,
   519→          rawCode: code,
   520→        },
   521→        { status: 422 }
   522→      );
   523→    }
   524→
   525→    if (STALE_CLIENT) {
   526→      // Raw SQL fallback for the stale-client case.
   527→      const id = makeId();
   528→      await rawCreateApp({
   529→        id,
   530→        name,
   531→        description,
   532→        category: safeCategory,
   533→        code,
   534→      });
   535→      return NextResponse.json({
   536→        ok: true,
   537→        app: { id, name, code },
   538→      });
   539→    }
   540→
   541→    const created = await db.generatedApp.create({
   542→      data: {
   543→        name,
   544→        description,
   545→        category: safeCategory,
   546→        code,
   547→      },
   548→    });
   549→
   550→    return NextResponse.json({
   551→      ok: true,
   552→      app: { id: created.id, name: created.name, code: created.code },
   553→    });
   554→  } catch (err) {
   555→    return NextResponse.json(
   556→      { ok: false, error: err instanceof Error ? err.message : "unknown" },
   557→      { status: 500 }
   558→    );
   559→  }
   560→}

   562→// DELETE — uninstall a generated app by id
   563→export async function DELETE(req: NextRequest) {
   564→  const { searchParams } = new URL(req.url);
   565→  const id = searchParams.get("id");
   566→
   567→  if (!id) {
   568→    return NextResponse.json(
   569→      { ok: false, error: "id query parameter is required" },
   570→      { status: 400 }
   571→    );
   572→  }
   573→
   574→  try {
   575→    if (STALE_CLIENT) {
   576→      const n = await rawDeleteApp(id);
   577→      if (n === 0) {
   578→        return NextResponse.json({ error: "App not found" }, { status: 404 });
   579→      }
   580→      return NextResponse.json({ ok: true });
   581→    }
   582→    await db.generatedApp.delete({ where: { id } });
   583→    return NextResponse.json({ ok: true });
   584→  } catch (err) {
   585→    return NextResponse.json(
   586→      { ok: false, error: err instanceof Error ? err.message : "unknown" },
